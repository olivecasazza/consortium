1//! # consortium-ray
2//!
3//! Ray job orchestration with nix-built environments.
4//!
5//! Supports both KubeRay (Kubernetes-native) and bare-metal Ray clusters.
6//! Uses nix to build hermetic job environments and the DAG executor for
7//! pipeline orchestration.
8//!
9//! All external commands run through the [`Executor`] abstraction, so the
10//! pipeline can be driven by a `ProcessExecutor` in production or a
11//! `ScriptedExecutor` in tests.
12
13pub mod error;
14pub mod tasks;

25/// Options controlling a ray job submission.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct RayOptions {
28    /// Whether to wait for the job to reach a terminal state before
29    /// returning. Defaults to `true`.
30    pub wait: bool,
31
32    /// Interval between `ray job status` polls while waiting. Defaults to
33    /// 10 seconds; test fixtures typically set this to ~0.
34    pub poll_interval: Duration,
35
36    /// Maximum time to wait for the job before failing the wait task.
37    /// `None` waits indefinitely.
38    pub timeout: Option<Duration>,
39}
40
41impl Default for RayOptions {
42    fn default() -> Self {
43        Self {
44            wait: true,
45            poll_interval: Duration::from_secs(10),
46            timeout: None,
47        }
48    }
49}

51/// Submit a ray job with a nix-built environment.
52///
53/// Builds the DAG `NixBuildRayEnvTask → RaySubmitTask (→ RayWaitTask when
54/// `opts.wait`)` and runs it to completion, with all commands executed
55/// through `exec`.
56pub fn submit_job(
57    exec: Arc<dyn Executor>,
58    config: &FleetConfig,
59    job_name: &str,
60    entrypoint: &str,
61    opts: &RayOptions,
62) -> Result<DagReport> {
63    let ray_config = config.ray_config.as_ref().ok_or(RayError::NoConfig)?;
64
65    let ctx = DagContext::new();
66    ctx.set_state("fleet_config", config.clone());
67    ctx.set_state("executor", exec);
68
69    let mut dag = DagBuilder::new();
70
71    // Build ray environment
72    let build_id = format!("build-ray-env:{}", job_name);
73    dag.add_task(
74        &build_id,
75        tasks::NixBuildRayEnvTask::new(job_name, &config.flake_uri),
76    );
77
78    // Submit job
79    let submit_id = format!("ray-submit:{}", job_name);
80    dag.add_task(
81        &submit_id,
82        tasks::RaySubmitTask {
83            job_name: job_name.to_string(),
84            entrypoint: entrypoint.to_string(),
85            head_address: ray_config.head_address.clone(),
86            dashboard_port: ray_config.dashboard_port,
87            working_dir: None,
88        },
89    );
90    dag.add_dep(&submit_id, &build_id);
91
92    // Wait (optional)
93    if opts.wait {
94        let wait_id = format!("ray-wait:{}", job_name);
95        dag.add_task(
96            &wait_id,
97            tasks::RayWaitTask {
98                job_name: job_name.to_string(),
99                head_address: ray_config.head_address.clone(),
100                dashboard_port: ray_config.dashboard_port,
101                poll_interval: opts.poll_interval,
102                timeout: opts.timeout,
103            },
104        );
105        dag.add_dep(&wait_id, &submit_id);
106    }
107
108    dag.error_policy(ErrorPolicy::FailFast);
109    dag.context(ctx);
110
111    let report = dag
112        .build()
113        .map_err(|e| RayError::Dag(e.to_string()))?
114        .run()
115        .map_err(|e| RayError::Dag(e.to_string()))?;
116
117    Ok(report)
118}
