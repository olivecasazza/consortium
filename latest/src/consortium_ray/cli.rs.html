1//! Lines printed by Ray's `cli_logger`. Without a TTY each line is a record,
2//! `<time>\t<LEVEL> cli.py:<line> -- <message>`, and the message may still be
3//! wrapped in one ANSI color.
4
5/// The message of a `cli.py` record logged at `level`, or of a plain (TTY) line.
6///
7/// Records from other files, such as job logs quoted in a status message, are
8/// ignored. Only one outer color is removed: escape codes inside the text
9/// remain, so splicing colored fragments cannot invent an identifier.
10pub(crate) fn message<'a>(line: &'a str, level: &str) -> Option<&'a str> {
11    let mut message = line;
12    if let Some((time, record)) = line.split_once('\t') {
13        let (location, text) = record
14            .strip_prefix(level)?
15            .strip_prefix(" cli.py:")?
16            .split_once(" -- ")?;
17        const TIMESTAMP: &[u8] = b"####-##-## ##:##:##,###";
18        if time.len() != TIMESTAMP.len()
19            || !time.bytes().zip(TIMESTAMP).all(|(c, &field)| match field {
20                b'#' => c.is_ascii_digit(),
21                _ => c == field,
22            })
23            || !decimal(location)
24        {
25            return None;
26        }
27        message = text;
28    }
29    if let Some(colored) = message.strip_prefix("\x1b[") {
30        let (style, text) = colored.split_once('m')?;
31        if !style.bytes().all(|b| b.is_ascii_digit() || b == b';') {
32            return None;
33        }
34        message = text
35            .strip_suffix("\x1b[39m")
36            .or_else(|| text.strip_suffix("\x1b[0m"))?;
37    }
38    Some(message)
39}

41pub(crate) fn decimal(value: &str) -> bool {
42    !value.is_empty() && value.bytes().all(|c| c.is_ascii_digit())
43}
