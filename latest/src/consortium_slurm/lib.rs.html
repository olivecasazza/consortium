1//! # consortium-slurm
2//!
3//! Slurm job orchestration with nix-built environments.
4//!
5//! Uses nix to build hermetic job environments, copies them to the submit
6//! node, then submits jobs with slurm. Supports arbitrary DAG pipelines
7//! (e.g., RNA-seq bioinformatics workflows) via DagBuilder.
8//!
9//! All external commands run through an injected
10//! `consortium_integration::exec::Executor`: production callers pass a
11//! `ProcessExecutor`, tests pass a `ScriptedExecutor`.

29/// Options for [`submit_job`].
30///
31/// The default exercises the full pipeline — build → copy → submit → wait —
32/// with no result collection.
33#[derive(Debug, Clone)]
34pub struct SubmitOptions {
35    /// Wait for the job to reach a terminal state (polls sacct).
36    /// Default: `true`.
37    pub wait: bool,
38    /// Literal output file path to `cat` on the submit node once the job
39    /// has finished (e.g. `"slurm-12345.out"`). Default: `None` (no
40    /// collection).
41    pub collect: Option<String>,
42    /// sacct poll interval used while waiting. Default: 10s.
43    pub poll_interval: Duration,
44}
45
46impl Default for SubmitOptions {
47    fn default() -> Self {
48        Self {
49            wait: true,
50            collect: None,
51            poll_interval: Duration::from_secs(10),
52        }
53    }
54}

56/// Submit a single slurm job with a nix-built environment.
57///
58/// Runs the pipeline `build-job-env → copy-to-submit → submit → wait →
59/// collect` as a DAG: the wait stage is included when
60/// [`SubmitOptions::wait`] is set, and the collect stage when
61/// [`SubmitOptions::collect`] is a literal file path — wired after `slurm-wait` when
62/// waiting is enabled, directly after `slurm-submit` otherwise. All commands
63/// run through `exec`.
64///
65/// Returns [`SlurmError::NoConfig`] before any command is issued when the
66/// fleet config has no slurm sub-config.
67pub fn submit_job(
68    exec: Arc<dyn Executor>,
69    config: &FleetConfig,
70    job_name: &str,
71    script: &str,
72    partition: Option<&str>,
73    opts: &SubmitOptions,
74) -> Result<DagReport> {
75    submit_job_with_timeout(exec, config, job_name, script, partition, opts, None)
76}
77
78/// Submit a job with an optional overall status-wait deadline.
79#[allow(clippy::too_many_arguments)]
80pub fn submit_job_with_timeout(
81    exec: Arc<dyn Executor>,
82    config: &FleetConfig,
83    job_name: &str,
84    script: &str,
85    partition: Option<&str>,
86    opts: &SubmitOptions,
87    timeout: Option<Duration>,
88) -> Result<DagReport> {
89    config.slurm_config.as_ref().ok_or(SlurmError::NoConfig)?;
90    let mut request = Request::default()
91        .with("name", job_name)
92        .with("script", script)
93        .with("wait", opts.wait.to_string());
94    if let Some(partition) = partition {
95        request = request.with("partition", partition);
96    }
97    if let Some(collect) = &opts.collect {
98        request = request.with("collect", collect);
99    }
100    if opts.wait {
101        request = request.with("poll-seconds", duration_text(opts.poll_interval));
102        if let Some(timeout) = timeout {
103            request = request.with("timeout", duration_text(timeout));
104        }
105    }
106    SlurmIntegration
107        .run_detailed(exec, config, &request)
108        .map(|result| result.dag)
109        .map_err(|error| SlurmError::Dag(error.to_string()))
110}

143    #[test]
144    fn typed_wait_preserves_subsecond_duration_precision() {
145        for (poll, timeout) in [
146            (Duration::from_nanos(1), Duration::from_nanos(3)),
147            (
148                Duration::from_nanos(1_234_567),
149                Duration::from_nanos(2_345_678),
150            ),
151            (Duration::from_millis(250), Duration::from_millis(600)),
152        ] {
153            let exec = running();
154            let report = submit_job_with_timeout(
155                exec.clone(),
156                &config(),
157                "test",
158                "run",
159                None,
160                &SubmitOptions {
161                    wait: true,
162                    collect: None,
163                    poll_interval: poll,
164                },
165                Some(timeout),
166            )
167            .unwrap();
168            assert_eq!(report.failed.len(), 1);
169            assert!(report
170                .failed
171                .values()
172                .any(|message| message.contains("timed out")));
173            assert_eq!(exec.monotonic_now(), timeout);
174            let sleeps: Vec<_> = exec
175                .effects()
176                .into_iter()
177                .filter_map(|effect| match effect {
178                    Effect::Sleep { duration } => Some(duration),
179                    _ => None,
180                })
181                .collect();
182            assert_eq!(sleeps.iter().copied().sum::<Duration>(), timeout);
183            assert!(sleeps
184                .iter()
185                .all(|duration| !duration.is_zero() && *duration <= poll));
186            let limits: Vec<_> = exec
187                .effects()
188                .into_iter()
189                .filter_map(|effect| match effect {
190                    Effect::Command {
191                        timeout: Some(limit),
192                        ..
193                    } => Some(limit),
194                    _ => None,
195                })
196                .collect();
197            assert_eq!(limits.first(), Some(&timeout));
198            assert!(limits.iter().all(|limit| !limit.is_zero()));
199        }
200    }
