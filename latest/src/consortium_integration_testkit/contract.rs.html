15/// A named stage of an integration's command pipeline.
16///
17/// # Marker discipline (read carefully)
18///
19/// `marker` is a plain substring matched against the *rendered* command line
20/// (`CommandSpec::render()`), which is what
21/// [`ScriptedExecutor`](consortium_integration::exec::ScriptedExecutor) rules
22/// match and what the harness asserts on. Markers must satisfy three rules:
23///
24/// 1. **Complete** — every command the integration issues for this phase
25///    contains the marker.
26/// 2. **Exclusive** — no command belonging to a *different* phase contains
27///    the marker. The harness enforces the static part of this (no two
28///    declared markers may be substrings of each other); the dynamic part
29///    (actual command lines) is the fixture's job.
30/// 3. **Quoting-safe** — ssh commands render with every remote token
31///    single-quoted: `ssh ... -l root host 'sbatch' '--job-name=x'`. A marker
32///    containing a space (e.g. `"sbatch --job-name"`) will *not* match that
33///    rendering. Use single tokens without spaces (`"sbatch"`) for ssh-wrapped
34///    phases. Space-containing markers (e.g. `"nix copy"`) are fine for
35///    phases that only ever run locally, where the rendered line is the raw
36///    argv joined by spaces.
37///
38/// `after` names phases that must precede this one. Phases must be declared
39/// in topological order: every `after` entry must name a phase declared
40/// *earlier* in the [`Contract::phases`] vector (the harness panics with a
41/// descriptive message otherwise).
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub struct Phase {
44    /// Human-readable phase name, unique within [`Contract::phases`].
45    pub name: &'static str,
46    /// Substring present in every rendered command line of this phase.
47    pub marker: &'static str,
48    /// Names of earlier phases that must complete before this one starts.
49    pub after: &'static [&'static str],
50}
51
52impl Phase {
53    /// Convenience constructor.
54    pub const fn new(
55        name: &'static str,
56        marker: &'static str,
57        after: &'static [&'static str],
58    ) -> Self {
59        Self {
60            name,
61            marker,
62            after,
63        }
64    }
65}

67/// One non-default option configuration the contract suite exercises.
68///
69/// The harness runs the integration with `options` on top of
70/// [`Contract::happy_executor`] and asserts:
71///
72/// - the run succeeds;
73/// - every phase named in `skipped_phases` was **not** invoked (matched by
74///   phase name against [`Contract::phases`]; unknown names fail the test);
75/// - every phase *not* listed was invoked.
76#[derive(Debug)]
77pub struct OptionVariant<Options> {
78    /// Variant name, used in failure messages.
79    pub name: &'static str,
80    /// Option values for this variant.
81    pub options: Options,
82    /// Names (not markers) of phases this variant intentionally skips.
83    pub skipped_phases: Vec<&'static str>,
84}
85
86impl<Options> OptionVariant<Options> {
87    /// Convenience constructor.
88    pub fn new(name: &'static str, options: Options, skipped_phases: Vec<&'static str>) -> Self {
89        Self {
90            name,
91            options,
92            skipped_phases,
93        }
94    }
95}

97/// Fixture for per-host partial failure: one host dies mid-pipeline while
98/// another must run to completion.
99///
100/// Semantics the harness asserts (when [`Contract::partial_failure`] returns
101/// `Some`):
102///
103/// - the outcome shows failure (either convention — see [`Contract::run`]);
104/// - for **every** declared phase, at least one invocation contains both the
105///   phase marker and `surviving_host_marker` (the surviving host completes
106///   its full chain);
107/// - no invocation contains both `blocked_phase_marker` and
108///   `failed_host_marker` (the failed host never reaches the blocked phase);
109/// - when the run returns `Ok(report)`, `report.success_count() >= 1 &&
110///   report.failure_count() >= 1`.
111///
112/// # Fixture requirements
113///
114/// - `executor` fails commands containing `failed_host_marker` at/after some
115///   point in the pipeline and lets everything else succeed. Put the failing
116///   [`Rule`](consortium_integration::exec::Rule) **before** the catch-all
117///   success rules — `ScriptedExecutor` is first-match-wins.
118/// - Every phase must issue at least one command per host that carries the
119///   host marker in its rendered line (make hostnames/addresses the host
120///   markers, and make sure they appear in every per-host command). If your
121///   integration has host-independent phases (e.g. one local build shared by
122///   all hosts), either give those commands a per-host form or return `None`
123///   and cover partial failure in integration-specific tests instead.
124/// - The integration must run with a continue-independent error policy for
125///   this scenario: the failed host's dependents are cancelled, the
126///   surviving host's chain runs to completion.
127pub struct PartialFailure {
128    /// Executor scripting the partial failure.
129    pub executor: ScriptedExecutor,
130    /// Substring identifying commands aimed at the host that fails.
131    pub failed_host_marker: &'static str,
132    /// Substring identifying commands aimed at the host that must survive.
133    pub surviving_host_marker: &'static str,
134    /// Phase marker that must never run for the failed host.
135    pub blocked_phase_marker: &'static str,
136}
137
138impl PartialFailure {
139    /// Convenience constructor.
140    pub fn new(
141        executor: ScriptedExecutor,
142        failed_host_marker: &'static str,
143        surviving_host_marker: &'static str,
144        blocked_phase_marker: &'static str,
145    ) -> Self {
146        Self {
147            executor,
148            failed_host_marker,
149            surviving_host_marker,
150            blocked_phase_marker,
151        }
152    }
153}

155/// The hook an integration implements to enroll in the abstract contract
156/// suite.
157///
158/// Implement this trait once per integration (typically in
159/// `tests/contract.rs`), then instantiate
160/// [`integration_contract_tests!`](crate::integration_contract_tests) to
161/// generate the seven uniform semantic tests. See the crate-level docs for a
162/// complete copy-pasteable example.
163///
164/// # Pinned-down semantics
165///
166/// - **Failure reporting conventions.** The harness tolerates both styles:
167///   `run` may return `Err(..)`, or it may return `Ok(report)` with
168///   `report.is_success() == false` and `report.failure_count() >= 1`. Pick
169///   one per situation and stay consistent; happy paths and option variants
170///   must always return `Ok` with a successful report.
171/// - **Missing configuration is an error, before any side effect.**
172///   `run(config_missing(), ..)` must return `Err` and must not have issued
173///   a single command.
174/// - **Planning is side-effect free.** [`Contract::task_descriptions`]
175///   receives an executor only so task construction can capture it; it must
176///   never call `exec` on it (the harness asserts zero invocations).
177/// - **`executor_failing_phase` receives a phase marker**, not a phase name.
178///   The returned executor must fail every command whose rendered line
179///   contains that marker (non-zero exit status with stderr is the realistic
180///   choice; [`ExecError`](consortium_integration::exec::ExecError)-style
181///   rule failures also work because the harness tolerates both outcome
182///   conventions) and let every other command succeed. Failing rules must be
183///   inserted before the catch-all success rules (first match wins).
184pub trait Contract {
185    /// The integration's run report.
186    type Report: IntegrationReport;
187    /// The integration's error type.
188    type Error: Debug;
189    /// Options modifying a run. Use `()` when the integration has no
190    /// variants (the harness only needs [`Default`], which `()` provides).
191    type Options: Default;
192
193    /// Integration name, used in every failure message (e.g. `"slurm"`).
194    fn name() -> &'static str;
195
196    /// A fleet config **missing this integration's sub-config** (e.g.
197    /// `slurm_config: None` for the slurm integration) but otherwise valid.
198    fn config_missing() -> FleetConfig;
199
200    /// A complete, valid fleet config with this integration's sub-config
201    /// present and at least one node.
202    fn config_complete() -> FleetConfig;
203
204    /// An executor where every command the integration issues succeeds with
205    /// realistic output (e.g. sbatch → `"Submitted batch job 12345\n"`).
206    /// A catch-all `.on("", ExecOutput::ok(..))` as the last rule guards
207    /// against unscripted commands failing the happy path spuriously.
208    fn happy_executor() -> ScriptedExecutor;
209
210    /// An executor where commands containing `phase_marker` fail (non-zero
211    /// exit + stderr recommended) and every other command succeeds.
212    /// `phase_marker` is always one of the markers from [`Contract::phases`].
213    fn executor_failing_phase(phase_marker: &str) -> ScriptedExecutor;
214
215    /// Invoke the integration's entry point.
216    fn run(
217        config: &FleetConfig,
218        exec: Arc<dyn Executor>,
219        opts: &Self::Options,
220    ) -> Result<Self::Report, Self::Error>;
221
222    /// The integration's pipeline phases, in topological order. See
223    /// [`Phase`] for the marker discipline the harness relies on.
224    fn phases() -> Vec<Phase>;
225
226    /// Non-default option variants to exercise. Default: none (the
227    /// option-variant test then passes trivially).
228    fn option_variants() -> Vec<OptionVariant<Self::Options>> {
229        Vec::new()
230    }
231
232    /// Per-host partial-failure fixture. Default: `None` (the harness skips
233    /// that test with a note). See [`PartialFailure`] for the semantics.
234    fn partial_failure() -> Option<PartialFailure> {
235        None
236    }
237
238    /// Construct the run's tasks for a representative small input and return
239    /// their human-readable descriptions, **without executing anything**.
240    ///
241    /// The `exec` parameter exists so integrations whose DAG tasks capture
242    /// the executor at construction time can build them; the harness asserts
243    /// afterwards that it recorded zero invocations. Descriptions must be
244    /// non-empty, single-line (no control characters) strings.
245    fn task_descriptions(config: &FleetConfig, exec: Arc<dyn Executor>) -> Vec<String>;
246}
