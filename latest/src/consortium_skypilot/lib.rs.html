1//! # consortium-skypilot
2//!
3//! SkyPilot multi-cloud job orchestration with nix-built environments.
4//!
5//! Launches cloud clusters via SkyPilot and tears them down afterwards,
6//! using nix for reproducible task environments. Every external command
7//! (`nix build`, `sky launch`, `sky down`) runs through an injected
8//! [`Executor`], which the DAG tasks share via the `"executor"` context
9//! state key — so the whole pipeline is testable with a
10//! [`ScriptedExecutor`](consortium_integration::exec::ScriptedExecutor)
11//! and needs neither a cloud account nor a nix installation.
12//!
13//! # Example
14//!
15//! ```
16//! use std::collections::HashMap;
17//! use std::sync::Arc;
18//!
19//! use consortium_integration::exec::{ExecOutput, Executor, ScriptedExecutor};
20//! use consortium_integration::fleet::{FleetConfig, SkypilotFleetConfig};
21//! use consortium_skypilot::{launch_task, SkyOptions};
22//!
23//! let exec: Arc<dyn Executor> = Arc::new(
24//!     ScriptedExecutor::new()
25//!         .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
26//!         .on("sky launch", ExecOutput::ok("Cluster launched\n"))
27//!         .on("sky down", ExecOutput::ok("Terminating cluster\n")),
28//! );
29//! let config = FleetConfig {
30//!     nodes: HashMap::new(),
31//!     builders: HashMap::new(),
32//!     flake_uri: ".".into(),
33//!     ansible_config: None,
34//!     slurm_config: None,
35//!     ray_config: None,
36//!     skypilot_config: Some(SkypilotFleetConfig {
37//!         cloud: "gcp".into(),
38//!         region: Some("us-central1".into()),
39//!         instance_type: None,
40//!     }),
41//! };
42//! let report = launch_task(exec, &config, "train", "resources: {}\n", &SkyOptions::default())?;
43//! assert!(report.failed.is_empty());
44//! # Ok::<(), consortium_skypilot::SkypilotError>(())
45//! ```

62/// Options for [`launch_task`].
63///
64/// The `Default` runs the full pipeline: build → launch → teardown.
65#[derive(Debug, Clone, Copy)]
66pub struct SkyOptions {
67    /// Tear the cluster down after it launches (`sky down`).
68    pub teardown: bool,
69}
70
71impl Default for SkyOptions {
72    fn default() -> Self {
73        Self { teardown: true }
74    }
75}

77/// Launch a SkyPilot task on a cloud cluster.
78///
79/// Pipeline: build the sky environment with nix → `sky launch` (the task
80/// yaml is written to a temp file so its path shows up in the rendered
81/// command line) → `sky down` when `opts.teardown`. Requires
82/// [`FleetConfig::skypilot_config`]; returns [`SkypilotError::NoConfig`]
83/// before issuing any command when it is absent.
84pub fn launch_task(
85    exec: Arc<dyn Executor>,
86    config: &FleetConfig,
87    cluster_name: &str,
88    task_yaml: &str,
89    opts: &SkyOptions,
90) -> Result<DagReport> {
91    config
92        .skypilot_config
93        .as_ref()
94        .ok_or(SkypilotError::NoConfig)?;
95    let request = Request::default()
96        .with("name", cluster_name)
97        .with("task", task_yaml)
98        .with("teardown", opts.teardown.to_string());
99    SkypilotIntegration
100        .run_detailed(exec, config, &request)
101        .map(|result| result.dag)
102        .map_err(|error| SkypilotError::Dag(error.to_string()))
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    use std::collections::HashMap;

110    use consortium_integration::exec::{ExecOutput, ScriptedExecutor};
111    use consortium_integration::fleet::SkypilotFleetConfig;

113    fn config(with_skypilot: bool) -> FleetConfig {
114        FleetConfig {
115            nodes: HashMap::new(),
116            builders: HashMap::new(),
117            flake_uri: ".".into(),
118            ansible_config: None,
119            slurm_config: None,
120            ray_config: None,
121            skypilot_config: with_skypilot.then(|| SkypilotFleetConfig {
122                cloud: "gcp".into(),
123                region: Some("us-central1".into()),
124                instance_type: None,
125            }),
126        }
127    }
128
129    #[test]
130    fn launch_task_errors_without_skypilot_config() {
131        let exec: Arc<dyn Executor> = Arc::new(ScriptedExecutor::new());
132        let err =
133            launch_task(exec, &config(false), "c", "yaml", &SkyOptions::default()).unwrap_err();
134        assert!(matches!(err, SkypilotError::NoConfig));
135    }
136
137    #[test]
138    fn launch_task_happy_path_orders_phases() {
139        let scripted = Arc::new(
140            ScriptedExecutor::new()
141                .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
142                .on("sky launch", ExecOutput::ok("Cluster launched\n"))
143                .on("sky down", ExecOutput::ok("Terminating cluster\n")),
144        );
145        let exec: Arc<dyn Executor> = scripted.clone();
146        let report = launch_task(
147            exec,
148            &config(true),
149            "train",
150            "resources: {}\n",
151            &SkyOptions::default(),
152        )
153        .unwrap();
154        assert!(report.failed.is_empty(), "failures: {:?}", report.failed);
155        let build = scripted.invocation_index_containing("nix build").unwrap();
156        let launch = scripted.invocation_index_containing("sky launch").unwrap();
157        let down = scripted.invocation_index_containing("sky down").unwrap();
158        assert!(build < launch && launch < down);
159    }
160
161    #[test]
162    fn launch_task_no_teardown_skips_down() {
163        let scripted = Arc::new(
164            ScriptedExecutor::new()
165                .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
166                .on("sky launch", ExecOutput::ok("Cluster launched\n")),
167        );
168        let exec: Arc<dyn Executor> = scripted.clone();
169        let report = launch_task(
170            exec,
171            &config(true),
172            "train",
173            "resources: {}\n",
174            &SkyOptions { teardown: false },
175        )
176        .unwrap();
177        assert!(report.failed.is_empty(), "failures: {:?}", report.failed);
178        scripted.assert_not_invoked_containing("sky down");
179    }
