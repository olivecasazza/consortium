1//! Production [`RoundExecutor`] implementations — drive real `nix copy`
2//! commands, runnable against actual nixlab hosts.
3//!
4//! [`NixCopyExecutor`] is the realistic counterpart to
5//! `consortium_fanout_sim::DeterministicExecutor`. The sim does
6//! `closure_size / bandwidth + latency` math; this one runs
7//! `nix copy --no-check-sigs --to ssh-ng://user@host store_path` through an
8//! [`Executor`] for every (src, tgt) edge in a round, in parallel via
9//! `std::thread`.
10//!
11//! ## Edge-source handling
12//!
13//! For an edge `(src, tgt)` with the cascade's seed at `seed_id`:
14//! - If `src == seed_id`: run `nix copy ...` LOCALLY (the host running
15//!   cascade-copy IS the seed)
16//! - Otherwise: SSH into `src` and run `nix copy ...` THERE (the source
17//!   forwards the closure it received in a prior round)
18//!
19//! ## Trust + signing
20//!
21//! `--no-check-sigs` is passed because closures built locally are NOT
22//! signed by a key the remote trusts. Trust boundary is the SSH
23//! connection itself (root-over-ssh deploy assumption — same as
24//! `consortium_nix::copy::copy_closure_with`).
25//!
26//! ## Failure mapping
27//!
28//! `nix copy` exit status maps to:
29//! - exit 0 → `Ok(elapsed_duration)`
30//! - non-zero with stderr containing "Connection refused"/"Permission
31//!   denied" → `CascadeError::SshHandshake` (permanent — no retry from
32//!   alt source will help if the tgt itself is unreachable)
33//! - non-zero otherwise → `CascadeError::Copy` (transient — retry from
34//!   alt source might succeed, e.g. if it was source-side bandwidth)
35

46/// Real-world `RoundExecutor` that drives `nix copy` over SSH.
47pub struct NixCopyExecutor {
48    /// Command executor every `nix copy` / `ssh` invocation runs through.
49    /// `ProcessExecutor` in production, `ScriptedExecutor` in tests.
50    pub exec: Arc<dyn Executor>,
51    /// NodeId → SSH address (e.g. `"root@hp01"` or `"olive@seir"`).
52    /// Seed node also has an entry here for symmetry, even though
53    /// edges originating from it run locally.
54    pub addrs: HashMap<NodeId, String>,
55    /// The store path being distributed, e.g. `/nix/store/xxx-foo-1.0`.
56    pub store_path: String,
57    /// NodeId of the seed — edges originating from it run via local
58    /// `nix copy`; all other src edges run via `ssh <src> 'nix copy …'`.
59    pub seed: NodeId,
60    /// Per-edge command timeout. Cascade halts the edge if the SSH
61    /// or `nix copy` hasn't returned by this point — typically the
62    /// remote is unreachable. Default 5 minutes.
63    pub timeout: Duration,
64}
65
66impl NixCopyExecutor {
67    /// An executor that runs every edge through `exec`.
68    pub fn new(
69        exec: Arc<dyn Executor>,
70        addrs: HashMap<NodeId, String>,
71        store_path: impl Into<String>,
72        seed: NodeId,
73    ) -> Self {
74        Self {
75            exec,
76            addrs,
77            store_path: store_path.into(),
78            seed,
79            timeout: Duration::from_secs(300),
80        }
81    }
82
83    /// An executor that spawns real subprocesses via [`ProcessExecutor`].
84    pub fn process(
85        addrs: HashMap<NodeId, String>,
86        store_path: impl Into<String>,
87        seed: NodeId,
88    ) -> Self {
89        Self::new(Arc::new(ProcessExecutor::new()), addrs, store_path, seed)
90    }
91
92    pub fn with_timeout(mut self, t: Duration) -> Self {
93        self.timeout = t;
94        self
95    }
96
97    /// Run a single edge: src copies the closure to tgt. Blocks until
98    /// the command completes or errors out.
99    ///
100    /// Returns `Ok(elapsed)` on success, `Err(CascadeError)` otherwise.
101    fn run_edge(&self, src: NodeId, tgt: NodeId) -> Result<Duration, CascadeError> {
102        let Some(tgt_addr) = self.addrs.get(&tgt) else {
103            return Err(CascadeError::Copy {
104                node: tgt,
105                stderr: format!("no SSH address registered for tgt {tgt}"),
106            });
107        };
108        let store_uri = format!("ssh-ng://{tgt_addr}");
109        let copy_args = [
110            "copy",
111            "--no-check-sigs",
112            "--to",
113            store_uri.as_str(),
114            self.store_path.as_str(),
115        ];
116
117        let started = Instant::now();
118        let spec = if src == self.seed {
119            // Local nix copy from the seed.
120            CommandSpec::new("nix").args(copy_args)
121        } else {
122            // SSH into src and have it run nix copy.
123            let Some(src_addr) = self.addrs.get(&src) else {
124                return Err(CascadeError::Copy {
125                    node: tgt,
126                    stderr: format!("no SSH address registered for src {src}"),
127                });
128            };
129            let (user, host) = split_ssh_addr(src_addr);
130            // accept-new overrides the DEFAULT_SSH_OPTS StrictHostKeyChecking=no
131            // (extra opts come after the defaults).
132            CommandSpec::new("nix")
133                .args(copy_args)
134                .ssh(SshTarget::new(user, host).extra_opt("-oStrictHostKeyChecking=accept-new"))
135        };
136
137        let result = self.exec.exec(&spec);
138        let elapsed = started.elapsed();
139        match result {
140            Ok(output) if output.success() => Ok(elapsed),
141            Ok(output) => Err(classify_copy_error(tgt, src, &output.stderr)),
142            Err(exec_err) => Err(CascadeError::Copy {
143                node: tgt,
144                stderr: format!("nix copy exec failed: {exec_err}"),
145            }),
146        }
147    }
148}

150impl RoundExecutor for NixCopyExecutor {
151    fn dispatch(
152        &self,
153        _nodes: &[CascadeNode],
154        edges: &[(NodeId, NodeId)],
155        _net: &NetworkProfile,
156    ) -> HashMap<(NodeId, NodeId), Result<Duration, CascadeError>> {
157        // Spawn one thread per edge; collect via channel so we don't
158        // need to box the closures or hold thread handles.
159        let (tx, rx) = mpsc::channel();
160        let n = edges.len();
161        thread::scope(|scope| {
162            for &(src, tgt) in edges {
163                let tx = tx.clone();
164                let me = &*self;
165                scope.spawn(move || {
166                    let outcome = me.run_edge(src, tgt);
167                    let _ = tx.send(((src, tgt), outcome));
168                });
169            }
170        });
171        drop(tx); // close the channel so the rx loop terminates
172        let mut out = HashMap::with_capacity(n);
173        while let Ok(item) = rx.try_recv() {
174            out.insert(item.0, item.1);
175        }
176        out
177    }
178}

180/// Map a non-zero `nix copy` stderr to the right `CascadeError` variant.
181/// Permanent vs transient distinction matters for orphan re-routing
182/// (see cascade.rs's `is_transient()` discussion).
183fn classify_copy_error(tgt: NodeId, src: NodeId, stderr: &str) -> CascadeError {
184    let lower = stderr.to_lowercase();
185    if lower.contains("connection refused")
186        || lower.contains("connection timed out")
187        || lower.contains("no route to host")
188        || lower.contains("permission denied")
189        || lower.contains("host key verification failed")
190    {
191        // Target host is permanently unreachable — orphan re-routing
192        // should kick in for any descendants in level-tree.
193        CascadeError::SshHandshake {
194            node: tgt,
195            parent: src,
196        }
197    } else {
198        // Default to transient — could be source-side bandwidth, a
199        // flaky relay, or a substituter being slow. Retry from an
200        // alternate source on the next round may succeed.
201        CascadeError::Copy {
202            node: tgt,
203            stderr: stderr.lines().take(5).collect::<Vec<_>>().join("\n"),
204        }
205    }
206}

208/// Split a `user@host` SSH address into its parts.
209///
210/// Fleet addresses are always `user@host`; a bare host falls back to
211/// `root` so hand-rolled inventories keep working.
212fn split_ssh_addr(addr: &str) -> (String, String) {
213    match addr.split_once('@') {
214        Some((user, host)) => (user.to_string(), host.to_string()),
215        None => ("root".to_string(), addr.to_string()),
216    }
217}

278    #[test]
279    fn dispatch_runs_edges_through_injected_executor() {
280        let scripted = Arc::new(ScriptedExecutor::new().on("", ExecOutput::ok("")));
281        let mut addrs = HashMap::new();
282        addrs.insert(NodeId(0), "root@seed".to_string());
283        addrs.insert(NodeId(1), "root@n1".to_string());
284        addrs.insert(NodeId(2), "root@n2".to_string());
285        let exec = NixCopyExecutor::new(scripted.clone(), addrs, "/nix/store/abc", NodeId(0));
286        let net = NetworkProfile::default();
287
288        // seed → n1 runs locally; n1 → n2 is relayed over ssh via n1.
289        let out = exec.dispatch(&[], &[(NodeId(0), NodeId(1)), (NodeId(1), NodeId(2))], &net);
290        assert!(out[&(NodeId(0), NodeId(1))].is_ok());
291        assert!(out[&(NodeId(1), NodeId(2))].is_ok());
292
293        // The seed edge is a bare local nix copy.
294        scripted.assert_invoked_containing(
295            "nix copy --no-check-sigs --to ssh-ng://root@n1 /nix/store/abc",
296        );
297        // The relayed edge is ssh-wrapped: ssh … -l root … n1 'nix' 'copy' …
298        let invocations = scripted.invocations();
299        let relay = invocations
300            .iter()
301            .find(|c| c.starts_with("ssh ") && c.contains("ssh-ng://root@n2"))
302            .expect("relayed edge should be ssh-wrapped");
303        assert!(relay.contains("-l root"), "{}", relay);
304        assert!(relay.contains("accept-new n1 "), "{}", relay);
305    }

307    #[test]
308    fn dispatch_maps_exec_error_to_copy_failure() {
309        let scripted = Arc::new(ScriptedExecutor::new().on_error("nix copy", "spawn blew up"));
310        let mut addrs = HashMap::new();
311        addrs.insert(NodeId(0), "root@seed".to_string());
312        addrs.insert(NodeId(1), "root@n1".to_string());
313        let exec = NixCopyExecutor::new(scripted, addrs, "/nix/store/abc", NodeId(0));
314        let net = NetworkProfile::default();
315
316        let out = exec.dispatch(&[], &[(NodeId(0), NodeId(1))], &net);
317        match &out[&(NodeId(0), NodeId(1))] {
318            Err(CascadeError::Copy { node, stderr }) => {
319                assert_eq!(*node, NodeId(1));
320                assert!(stderr.contains("spawn blew up"), "{}", stderr);
321            }
322            other => panic!("expected transient Copy error, got: {:?}", other),
323        }
324    }
