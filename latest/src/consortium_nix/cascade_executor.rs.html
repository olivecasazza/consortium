1//! Production [`RoundExecutor`] implementations — drive real `nix copy`
2//! commands, runnable against actual nixlab hosts.
3//!
4//! [`NixCopyExecutor`] is the realistic counterpart to
5//! `consortium_fanout_sim::DeterministicExecutor`. The sim does
6//! `closure_size / bandwidth + latency` math; this one runs
7//! `nix copy --no-check-sigs --to ssh-ng://user@host store_path` through an
8//! [`Executor`] for every (src, tgt) edge in a round, in parallel via
9//! `std::thread`.
10//!
11//! ## Edge-source handling
12//!
13//! For an edge `(src, tgt)` with the cascade's seed at `seed_id`:
14//! - If `src == seed_id`: run `nix copy ...` LOCALLY (the host running
15//!   cascade-copy IS the seed)
16//! - Otherwise: SSH into `src` and run `nix copy ...` THERE (the source
17//!   forwards the closure it received in a prior round)
18//!
19//! ## Trust + signing
20//!
21//! `--no-check-sigs` is passed because closures built locally are NOT
22//! signed by a key the remote trusts. Trust boundary is the SSH
23//! connection itself (root-over-ssh deploy assumption — same as
24//! `consortium_nix::copy::copy_closure_with`).
25//!
26//! ## Failure mapping
27//!
28//! `nix copy` exit status maps to:
29//! - exit 0 → `Ok(elapsed_duration)`
30//! - non-zero with stderr containing "Connection refused"/"Permission
31//!   denied" → `CascadeError::SshHandshake` (permanent — no retry from
32//!   alt source will help if the tgt itself is unreachable)
33//! - non-zero otherwise → `CascadeError::Copy` (transient — retry from
34//!   alt source might succeed, e.g. if it was source-side bandwidth)
35

46/// Real-world `RoundExecutor` that drives `nix copy` over SSH.
47pub struct NixCopyExecutor {
48    /// Command executor every `nix copy` / `ssh` invocation runs through.
49    /// `ProcessExecutor` in production, `ScriptedExecutor` in tests.
50    pub exec: Arc<dyn Executor>,
51    /// NodeId → SSH address (e.g. `"root@hp01"` or `"olive@seir"`).
52    /// Seed node also has an entry here for symmetry, even though
53    /// edges originating from it run locally.
54    pub addrs: HashMap<NodeId, String>,
55    /// The store path being distributed, e.g. `/nix/store/xxx-foo-1.0`.
56    pub store_path: String,
57    /// NodeId of the seed — edges originating from it run via local
58    /// `nix copy`; all other src edges run via `ssh <src> 'nix copy …'`.
59    pub seed: NodeId,
60    /// Per-edge command timeout. When set, an edge whose SSH or `nix copy`
61    /// hasn't returned by then is killed. `None` (the default) lets a copy
62    /// run to completion.
63    pub timeout: Option<Duration>,
64}

66impl NixCopyExecutor {
67    /// An executor that runs every edge through `exec`.
68    pub fn new(
69        exec: Arc<dyn Executor>,
70        addrs: HashMap<NodeId, String>,
71        store_path: impl Into<String>,
72        seed: NodeId,
73    ) -> Self {
74        Self {
75            exec,
76            addrs,
77            store_path: store_path.into(),
78            seed,
79            timeout: None,
80        }
81    }

83    /// An executor that spawns real subprocesses via [`ProcessExecutor`].
84    pub fn process(
85        addrs: HashMap<NodeId, String>,
86        store_path: impl Into<String>,
87        seed: NodeId,
88    ) -> Self {
89        Self::new(Arc::new(ProcessExecutor::new()), addrs, store_path, seed)
90    }
91
92    pub fn with_timeout(mut self, t: Duration) -> Self {
93        self.timeout = Some(t);
94        self
95    }

97    /// Run a single edge: src copies the closure to tgt. Blocks until
98    /// the command completes or errors out.
99    ///
100    /// Returns `Ok(elapsed)` on success, `Err(CascadeError)` otherwise.
101    fn run_edge(&self, src: NodeId, tgt: NodeId) -> Result<Duration, CascadeError> {
102        let Some(tgt_addr) = self.addrs.get(&tgt) else {
103            return Err(CascadeError::Copy {
104                node: tgt,
105                stderr: format!("no SSH address registered for tgt {tgt}"),
106            });
107        };
108        let store_uri = format!("ssh-ng://{tgt_addr}");
109        let copy_args = [
110            "copy",
111            "--no-check-sigs",
112            "--to",
113            store_uri.as_str(),
114            self.store_path.as_str(),
115        ];
116
117        let started = self.exec.monotonic_now();
118        let spec = if src == self.seed {
119            // Local nix copy from the seed.
120            CommandSpec::new("nix").args(copy_args)
121        } else {
122            // SSH into src and have it run nix copy.
123            let Some(src_addr) = self.addrs.get(&src) else {
124                return Err(CascadeError::Copy {
125                    node: tgt,
126                    stderr: format!("no SSH address registered for src {src}"),
127                });
128            };
129            let (user, host, port) = split_ssh_addr(src_addr);
130            // accept-new overrides the DEFAULT_SSH_OPTS StrictHostKeyChecking=no
131            // (extra opts come after the defaults).
132            CommandSpec::new("nix").args(copy_args).ssh(match port {
133                // The port belongs on the command line; leaving it on the
134                // host makes ssh resolve "host:port" as a single name.
135                Some(p) => SshTarget::new(user, host)
136                    .extra_opt("-oStrictHostKeyChecking=accept-new")
137                    .port(p),
138                None => SshTarget::new(user, host).extra_opt("-oStrictHostKeyChecking=accept-new"),
139            })
140        };
141
142        let result = match self.timeout {
143            Some(limit) => self.exec.exec_timeout(&spec, limit),
144            None => self.exec.exec(&spec),
145        };
146        let elapsed = self.exec.monotonic_now().saturating_sub(started);
147        match result {
148            Ok(output) if output.success() => Ok(elapsed),
149            Ok(output) => Err(classify_copy_error(tgt, src, &output.stderr)),
150            Err(exec_err) => Err(CascadeError::Copy {
151                node: tgt,
152                stderr: format!("nix copy exec failed: {exec_err}"),
153            }),
154        }
155    }
156}

158impl RoundExecutor for NixCopyExecutor {
159    fn wall_now(&self) -> std::time::SystemTime {
160        self.exec.wall_now()
161    }
162
163    fn dispatch(
164        &self,
165        _nodes: &[CascadeNode],
166        edges: &[(NodeId, NodeId)],
167        _net: &NetworkProfile,
168    ) -> HashMap<(NodeId, NodeId), Result<Duration, CascadeError>> {
169        // Spawn one thread per edge; collect via channel so we don't
170        // need to box the closures or hold thread handles.
171        let (tx, rx) = mpsc::channel();
172        let n = edges.len();
173        thread::scope(|scope| {
174            for &(src, tgt) in edges {
175                let tx = tx.clone();
176                let me = &*self;
177                scope.spawn(move || {
178                    let outcome = me.run_edge(src, tgt);
179                    let _ = tx.send(((src, tgt), outcome));
180                });
181            }
182        });
183        drop(tx); // close the channel so the rx loop terminates
184        let mut out = HashMap::with_capacity(n);
185        while let Ok(item) = rx.try_recv() {
186            out.insert(item.0, item.1);
187        }
188        out
189    }
190}

192/// Map a non-zero `nix copy` stderr to the right `CascadeError` variant.
193/// Permanent vs transient distinction matters for orphan re-routing
194/// (see cascade.rs's `is_transient()` discussion).
195fn classify_copy_error(tgt: NodeId, src: NodeId, stderr: &str) -> CascadeError {
196    let lower = stderr.to_lowercase();
197    if lower.contains("connection refused")
198        || lower.contains("connection timed out")
199        || lower.contains("no route to host")
200        || lower.contains("permission denied")
201        || lower.contains("host key verification failed")
202    {
203        // Target host is permanently unreachable — orphan re-routing
204        // should kick in for any descendants in level-tree.
205        CascadeError::SshHandshake {
206            node: tgt,
207            parent: src,
208        }
209    } else {
210        // Default to transient — could be source-side bandwidth, a
211        // flaky relay, or a substituter being slow. Retry from an
212        // alternate source on the next round may succeed.
213        CascadeError::Copy {
214            node: tgt,
215            stderr: stderr.lines().take(5).collect::<Vec<_>>().join("\n"),
216        }
217    }
218}

220/// Split a `user@host[:port]` SSH address into its parts.
221///
222/// Fleet addresses are normally `user@host`; a bare host falls back to `root`
223/// so hand-rolled inventories keep working.
224///
225/// A port in the address is returned separately rather than left on the host.
226/// `ssh` reads `10.0.2.2:22202` as one hostname and fails with "Could not
227/// resolve hostname", so an inventory that puts the port in the address — which
228/// is how a port-forwarded fleet addresses its guests — silently loses every
229/// edge that goes through here.
230fn split_ssh_addr(addr: &str) -> (String, String, Option<u16>) {
231    let (user, host) = match addr.split_once('@') {
232        Some((user, host)) => (user.to_string(), host.to_string()),
233        None => ("root".to_string(), addr.to_string()),
234    };
235    if let Some(rest) = host.strip_prefix('[') {
236        if let Some((literal, suffix)) = rest.split_once(']') {
237            let port = suffix
238                .strip_prefix(':')
239                .and_then(|value| value.parse::<u16>().ok());
240            return (user, literal.to_owned(), port);
241        }
242    }
243    // An IPv6 literal has several colons, and its last group is often all
244    // digits (`fe80::1`). Only a single colon can introduce a port.
245    if host.matches(':').count() != 1 {
246        return (user, host, None);
247    }
248    match host.rsplit_once(':') {
249        // Only a trailing all-digits port counts.
250        Some((host, port)) if !port.is_empty() && port.chars().all(|c| c.is_ascii_digit()) => {
251            match port.parse::<u16>() {
252                Ok(port) => (user, host.to_string(), Some(port)),
253                Err(_) => (user, host.to_string(), None),
254            }
255        }
256        _ => (user, host, None),
257    }
258}

352    #[test]
353    fn dispatch_runs_edges_through_injected_executor() {
354        let scripted = Arc::new(ScriptedExecutor::new().on("", ExecOutput::ok("")));
355        let mut addrs = HashMap::new();
356        addrs.insert(NodeId(0), "root@seed".to_string());
357        addrs.insert(NodeId(1), "root@n1".to_string());
358        addrs.insert(NodeId(2), "root@n2".to_string());
359        let exec = NixCopyExecutor::new(scripted.clone(), addrs, "/nix/store/abc", NodeId(0));
360        let net = NetworkProfile::default();
361
362        // seed → n1 runs locally; n1 → n2 is relayed over ssh via n1.
363        let out = exec.dispatch(&[], &[(NodeId(0), NodeId(1)), (NodeId(1), NodeId(2))], &net);
364        assert!(out[&(NodeId(0), NodeId(1))].is_ok());
365        assert!(out[&(NodeId(1), NodeId(2))].is_ok());
366
367        // The seed edge is a bare local nix copy.
368        scripted.assert_invoked_containing(
369            "nix copy --no-check-sigs --to ssh-ng://root@n1 /nix/store/abc",
370        );
371        // The relayed edge is ssh-wrapped: ssh … -l root … n1 'nix' 'copy' …
372        let invocations = scripted.invocations();
373        let relay = invocations
374            .iter()
375            .find(|c| c.starts_with("ssh ") && c.contains("ssh-ng://root@n2"))
376            .expect("relayed edge should be ssh-wrapped");
377        assert!(relay.contains("-l root"), "{}", relay);
378        assert!(relay.contains("accept-new -- n1 "), "{}", relay);
379    }
