1//! Cascade closure-distribution primitive.
2//!
3//! Replaces source-to-each-target serial copy with a log-N peer-to-peer
4//! cascade. Once a node has the closure, it joins the source pool and
5//! serves the next round of targets. Errors aggregate up the cascade
6//! tree so the user-facing error names every failed node with its
7//! parent path intact.
8//!
9//! ## Architecture
10//!
11//! - [`CascadeNode`] holds per-node identity + parent/children topology
12//!   built up as the cascade runs.
13//! - [`NetworkProfile`] describes per-edge bandwidth/latency/partition
14//!   state. Empty (default) means uniform, well-connected.
15//! - [`CascadeStrategy`] is the pluggable decision surface: given the
16//!   current cascade state and the network profile, decide which
17//!   `(src, tgt)` edges to fire this round. [`Log2FanOut`] is the
18//!   baseline; [`crate::cascade_strategies`] hosts cost-aware variants
19//!   that pull in petgraph.
20//! - [`RoundExecutor`] is how we actually fire those edges. Production
21//!   wires it to `nix copy` (one subprocess per edge, std::thread for
22//!   in-round parallelism). Sim wires it to deterministic
23//!   bandwidth-driven `madsim::time::sleep`.
24//! - [`run_cascade`] is the coordinator loop: ask strategy → dispatch
25//!   → record outcomes → repeat until convergence or no further
26//!   progress.
27//!
28//! ## Error semantics
29//!
30//! When a copy fails, the target's parent (the node that tried to
31//! serve it) records the error. At the end of each round those errors
32//! bubble one level up: the target's parent's own parent receives a
33//! [`CascadeError::SubtreeAggregate`] keyed by the failing branch.
34//! Sibling failures merge into the same aggregate, so the user-facing
35//! error contains the full failed-subtree shape — not "first error
36//! wins."

107/// Which source-pool traversal a fanout round uses when pairing
108/// sources to targets — a fanout dimension, orthogonal to the strategy
109/// registry (an ordering of edges within a round, not a topology).
110///
111/// It reaches the pairing through the strategy, not through
112/// [`CascadeState`] or [`NetworkProfile`]: both are public structs
113/// exhaustively constructible downstream, so a new field on either is
114/// a semver-major break (the gate rejects it) — while a new strategy
115/// type implementing the unchanged [`CascadeStrategy`] trait is purely
116/// additive. See [`OrderedFanout`].
117#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
118#[non_exhaustive]
119pub enum FanoutOrder {
120    /// The historical pairing: sources in id order, targets in id
121    /// order, greedily paired in that order. Byte-for-byte what
122    /// [`Log2FanOut`] has always done.
123    #[default]
124    Default,
125    /// Shallowest sources first: nodes nearest a seeded root serve
126    /// before their own descendants, so the relay tree stays shallow.
127    /// Ties break by id.
128    Bfs,
129    /// Deepest sources first: the deepest branch is extended before
130    /// moving sideways. Ties break by id.
131    Dfs,
132}
133
134/// One registered traversal order: its canonical name, friendly
135/// aliases, the value, and a one-line description. Everything that
136/// names an order reads this table — [`parse_fanout_order`], the
137/// unknown-name error, and the discovery listing.
138pub struct OrderSpec {
139    /// The name diagnostics print.
140    pub canonical: &'static str,
141    /// Short friendly spellings accepted alongside the canonical name.
142    pub aliases: &'static [&'static str],
143    /// The value a [`NetworkProfile`] carries.
144    pub order: FanoutOrder,
145    /// One-line description for discovery output.
146    pub description: &'static str,
147}
148
149/// Every fanout traversal order, in display order. `default` is first
150/// because it is what runs when `--order` is omitted.
151pub const FANOUT_ORDERS: &[OrderSpec] = &[
152    OrderSpec {
153        canonical: "default",
154        aliases: &["id"],
155        order: FanoutOrder::Default,
156        description: "sources in id order (the historical pairing)",
157    },
158    OrderSpec {
159        canonical: "bfs",
160        aliases: &["breadth-first"],
161        order: FanoutOrder::Bfs,
162        description: "shallowest sources first (nearest the root)",
163    },
164    OrderSpec {
165        canonical: "dfs",
166        aliases: &["depth-first"],
167        order: FanoutOrder::Dfs,
168        description: "deepest branch first",
169    },
170];
171
172/// Resolve a traversal order by canonical name or friendly alias.
173pub fn find_fanout_order(name: &str) -> Option<&'static OrderSpec> {
174    FANOUT_ORDERS
175        .iter()
176        .find(|o| o.canonical == name || o.aliases.contains(&name))
177}
178
179/// Every registered order as "`canonical` (aliases: a, b)", in table
180/// order. Generated from [`FANOUT_ORDERS`] — this is what the
181/// unknown-name error prints and what the discovery listing reads, so
182/// the two cannot drift apart.
183pub fn fanout_order_catalog() -> String {
184    FANOUT_ORDERS
185        .iter()
186        .map(|o| {
187            if o.aliases.is_empty() {
188                o.canonical.to_string()
189            } else {
190                format!("{} (aliases: {})", o.canonical, o.aliases.join(", "))
191            }
192        })
193        .collect::<Vec<_>>()
194        .join(", ")
195}

216/// Resolve `name` through [`FANOUT_ORDERS`]. Unknown or empty names

231/// Per-node link capacity. Absent → unbounded (degenerates to per-edge
232/// bandwidth only, matching the simple model). Present → contention
233/// math kicks in: a source serving N targets in one round shares its
234/// uplink N ways.
235///
236/// Users opt into realism by *describing more of the network*, not by
237/// flipping a "mode" — the executor reads `effective_bandwidth` which
238/// always returns the right number whether contention is modeled or not.
239#[derive(Debug, Clone, Copy)]
240pub struct NodeSpec {
241    /// Outbound capacity in bytes/sec.
242    pub uplink: u64,
243    /// Inbound capacity in bytes/sec.
244    pub downlink: u64,
245}
246
247impl NodeSpec {
248    pub fn symmetric(bytes_sec: u64) -> Self {
249        Self {
250            uplink: bytes_sec,
251            downlink: bytes_sec,
252        }
253    }
254}
255
256/// Network properties: per-edge specs + optional per-node link
257/// capacities. Empty fields fall back to caller-supplied defaults at
258/// lookup time. Strategies that don't care about the network simply
259/// ignore this struct.
260#[derive(Debug, Default, Clone)]
261pub struct NetworkProfile {
262    /// bytes/second per directed edge
263    pub bandwidth: HashMap<(NodeId, NodeId), u64>,
264    /// per-edge latency
265    pub latency: HashMap<(NodeId, NodeId), Duration>,
266    /// edges that cannot pass traffic at all
267    pub partitions: HashSet<(NodeId, NodeId)>,
268    /// per-node link capacities. Absent → no contention modeling for
269    /// that node (treated as having infinite uplink/downlink).
270    pub nodes: HashMap<NodeId, NodeSpec>,
271}
272
273impl NetworkProfile {
274    pub fn bandwidth_of(&self, src: NodeId, tgt: NodeId, default: u64) -> u64 {
275        self.bandwidth.get(&(src, tgt)).copied().unwrap_or(default)
276    }
277
278    pub fn latency_of(&self, src: NodeId, tgt: NodeId, default: Duration) -> Duration {
279        self.latency.get(&(src, tgt)).copied().unwrap_or(default)
280    }
281
282    pub fn is_partitioned(&self, src: NodeId, tgt: NodeId) -> bool {
283        self.partitions.contains(&(src, tgt))
284    }
285
286    /// Compute the effective bandwidth available on `(src, tgt)` given
287    /// how many edges are simultaneously fanning out from `src` and
288    /// fanning in to `tgt` this round.
289    ///
290    /// Returns `min(edge_bw, src.uplink/src_out_count, tgt.downlink/tgt_in_count)`.
291    /// If a node has no `NodeSpec`, its term is `u64::MAX` (effectively
292    /// no constraint), so the math degenerates to `edge_bw` — same as
293    /// the pre-contention behavior.
294    ///
295    /// Counts are clamped to `>=1` to avoid divide-by-zero (a count of 0
296    /// would mean this edge isn't actually scheduled, but defensively
297    /// we don't trust the caller).
298    pub fn effective_bandwidth(
299        &self,
300        src: NodeId,
301        tgt: NodeId,
302        src_out_count: u64,
303        tgt_in_count: u64,
304        default_edge_bw: u64,
305    ) -> u64 {
306        let edge_bw = self.bandwidth_of(src, tgt, default_edge_bw);
307        let src_share = self
308            .nodes
309            .get(&src)
310            .map(|s| s.uplink / src_out_count.max(1))
311            .unwrap_or(u64::MAX);
312        let tgt_share = self
313            .nodes
314            .get(&tgt)
315            .map(|t| t.downlink / tgt_in_count.max(1))
316            .unwrap_or(u64::MAX);
317        edge_bw.min(src_share).min(tgt_share)
318    }
319}

325/// What a strategy returns for a single round.
326#[derive(Debug, Clone, Default)]
327pub struct CascadePlan {
328    pub round: u32,
329    /// `(src_with_closure, tgt_without_closure)` edges to fire this round.

333}

335/// Read-only snapshot the strategy inspects to plan the next round.
336#[derive(Debug)]
337pub struct CascadeState<'a> {
338    pub nodes: &'a [CascadeNode],
339    pub has_closure: &'a HashSet<NodeId>,
340    pub round: u32,
341    /// Edges already attempted in prior rounds (success or failure).
342    /// A strategy must not re-issue these.
343    pub attempted: &'a HashSet<(NodeId, NodeId)>,
344    /// Nodes that have failed at least once and should be excluded from
345    /// further attempts (set by the coordinator after a failed round).
346    pub failed_nodes: &'a HashSet<NodeId>,
347}

349/// Plug-in cascade decision logic.
350///
351/// Per-strategy convergence estimates (`expected_rounds`) live as
352/// inherent methods on each concrete strategy (e.g.
353/// [`Log2FanOut::expected_rounds`]) — they are strategy-specific

358    fn name(&self) -> &'static str;
359}

361// ============================================================================
362// Error tree
363// ============================================================================
364
365/// Errors aggregate up the cascade tree: a parent collects its
366/// children's errors via [`CascadeError::merge`], producing a
367/// [`SubtreeAggregate`](CascadeError::SubtreeAggregate) the next level
368/// up consumes.
369#[derive(Debug, Clone, Error)]
370pub enum CascadeError {
371    #[error("copy to {node} failed: {stderr}")]
372    Copy { node: NodeId, stderr: String },
373
374    #[error("ssh handshake to {node} failed (from parent {parent})")]
375    SshHandshake { node: NodeId, parent: NodeId },
376
377    #[error("activation on {node} failed at stage {stage}")]
378    Activation { node: NodeId, stage: &'static str },
379
380    #[error("partition: cannot reach {tgt} from {src}")]
381    Partitioned { src: NodeId, tgt: NodeId },
382
383    #[error("subtree rooted at {node}: {} failure(s) below", errors.len())]
384    SubtreeAggregate {
385        node: NodeId,
386        errors: Vec<CascadeError>,
387    },
388}
389
390impl CascadeError {
391    /// Whether the error is "transient" — a single edge failure that
392    /// might succeed on retry from a different source. The coordinator
393    /// uses this to decide whether to mark a target as permanently
394    /// failed (dead-letter) or leave it open for retry:
395    ///
396    /// - `Copy` → transient (network glitch, transient bandwidth issue)
397    /// - `SshHandshake`, `Activation`, `Partitioned` → permanent
398    ///   (target node is dead / unreachable / activation failure)
399    /// - `SubtreeAggregate` → never used as a per-edge outcome
400    pub fn is_transient(&self) -> bool {
401        matches!(self, CascadeError::Copy { .. })
402    }
403
404    /// Fold a parent's children's errors into a SubtreeAggregate.
405    pub fn merge(parent: NodeId, errors: Vec<CascadeError>) -> Self {
406        Self::SubtreeAggregate {
407            node: parent,
408            errors,
409        }
410    }
411
412    /// Every leaf NodeId that contributed to this error tree.
413    pub fn affected_nodes(&self) -> Vec<NodeId> {
414        let mut acc = Vec::new();
415        self.collect_affected(&mut acc);
416        acc
417    }
418
419    fn collect_affected(&self, acc: &mut Vec<NodeId>) {
420        match self {
421            CascadeError::Copy { node, .. }
422            | CascadeError::SshHandshake { node, .. }
423            | CascadeError::Activation { node, .. } => acc.push(*node),
424            CascadeError::Partitioned { tgt, .. } => acc.push(*tgt),
425            CascadeError::SubtreeAggregate { errors, .. } => {
426                for e in errors {
427                    e.collect_affected(acc);
428                }
429            }
430        }
431    }
432
433    /// Walk the error tree, yielding (depth, error) for each non-aggregate.
434    pub fn walk_leaves(&self, mut f: impl FnMut(usize, &CascadeError)) {
435        fn go(err: &CascadeError, depth: usize, f: &mut dyn FnMut(usize, &CascadeError)) {
436            match err {
437                CascadeError::SubtreeAggregate { errors, .. } => {
438                    for e in errors {
439                        go(e, depth + 1, f);
440                    }
441                }
442                _ => f(depth, err),
443            }
444        }
445        go(self, 0, &mut f);
446    }
447}

650// NetworkBuilder — describe the network conversationally
651// ============================================================================
652
653/// Builder for [`NetworkProfile`]. Each method describes one facet of
654/// the network's physics; calling more methods adds more realism.
655///
656/// Realism is fully opt-in: a `NetworkBuilder::new().build()` produces
657/// an empty profile (no contention, no partitions, no per-edge specs)
658/// — same as today's default. Adding `.uplinks(...)` engages contention
659/// math automatically.
660///
661/// # Example
662///
663/// ```ignore
664/// let net = NetworkBuilder::new()
665///     .partitions([(NodeId(0), NodeId(7))])
666///     .uplinks_uniform(1024 * 1024 * 1024)  // 1 Gbps each
667///     .build();
668/// ```
669#[derive(Debug, Default)]
670pub struct NetworkBuilder {
671    profile: NetworkProfile,
672}
673
674impl NetworkBuilder {
675    pub fn new() -> Self {
676        Self::default()
677    }
678
679    /// Set per-edge bandwidth from an arbitrary iterator.
680    pub fn bandwidth<I>(mut self, edges: I) -> Self
681    where
682        I: IntoIterator<Item = ((NodeId, NodeId), u64)>,
683    {
684        self.profile.bandwidth.extend(edges);
685        self
686    }
687
688    /// Set per-edge latency from an arbitrary iterator.
689    pub fn latency<I>(mut self, edges: I) -> Self
690    where
691        I: IntoIterator<Item = ((NodeId, NodeId), Duration)>,
692    {

741/// Run a cascade end-to-end.
742///
743/// The cascade halts when any of:
744/// - every reachable node has the closure (success)
745/// - the strategy returns an empty plan (no further progress possible)
746/// - `max_rounds` is reached (sanity bound)
747///
748/// `seeded` is the set of nodes that already hold the closure at start.
749/// Must be non-empty unless `nodes` is also empty.
750///
751/// Prefer the [`Cascade`] builder for new call sites — this raw fn is
752/// kept for backwards compatibility and for the builder's `.run()` to
753/// delegate to.
754pub fn run_cascade(
755    nodes: Vec<CascadeNode>,
756    seeded: HashSet<NodeId>,
757    net: NetworkProfile,
758    strategy: &dyn CascadeStrategy,
759    executor: &dyn RoundExecutor,
760    max_rounds: u32,
761    trace: Option<&dyn TraceSink>,
762) -> CascadeResult {
763    run_cascade_with_events(
764        nodes, seeded, net, strategy, executor, max_rounds, trace, None,
765    )
766}

768/// Like [`run_cascade`] but also emits fine-grained
769/// [`CascadeEvent`](crate::cascade_events::CascadeEvent)s through
770/// `events`. Use this for live tracing, JSONL persistence, or any
771/// consumer that wants per-edge timing rather than per-round
772/// aggregates.
773///
774/// `trace` and `events` are independent — passing both is fine, useful
775/// for migrating from the snapshot model to the event model
776/// incrementally.
777#[allow(clippy::too_many_arguments)]
778pub fn run_cascade_with_events(
779    mut nodes: Vec<CascadeNode>,
780    seeded: HashSet<NodeId>,
781    net: NetworkProfile,
782    strategy: &dyn CascadeStrategy,
783    executor: &dyn RoundExecutor,
784    max_rounds: u32,
785    trace: Option<&dyn TraceSink>,
786    events: Option<&dyn crate::cascade_events::EventSink>,
787) -> CascadeResult {
788    use crate::cascade_events::{CascadeEvent, Edge};
789    use std::time::SystemTime;
790
791    let strategy_name = strategy.name().to_string();
792
793    if let Some(sink) = events {
794        let mut seeded_vec: Vec<NodeId> = seeded.iter().copied().collect();
795        seeded_vec.sort();
796        sink.emit(&CascadeEvent::Started {
797            n_nodes: nodes.len() as u32,
798            seeded: seeded_vec,
799            strategy: strategy_name.clone(),
800            at: SystemTime::now(),
801        });
802    }
803
804    let mut has_closure = seeded.clone();
805    let mut attempted: HashSet<(NodeId, NodeId)> = HashSet::new();
806    // failed_nodes contains targets that hit a PERMANENT error
807    // (Activation, SshHandshake, Partitioned — i.e. the node is
808    // genuinely dead/unreachable). Transient Copy errors do NOT mark
809    // failed_nodes; the strategy retries from an alternate source.
810    let mut failed_nodes: HashSet<NodeId> = HashSet::new();
811    let mut round_durations: Vec<Duration> = Vec::new();
812
813    // per-parent error bucket: when round k completes, the parent of
814    // each failed target collects that error here. At end of round
815    // these get merged into SubtreeAggregates and bubbled one level up
816    // by attaching them to the *parent's* parent's bucket next round.
817    let mut pending_errors: HashMap<NodeId, Vec<CascadeError>> = HashMap::new();
818    // root-level (no-parent) errors — these are pre-seeded sources or
819    // the initial seed. Anything that bubbles all the way up lands here.
820    let mut root_errors: Vec<CascadeError> = Vec::new();
821
822    let mut round = 0u32;
823    while round < max_rounds {
824        let state = CascadeState {
825            nodes: &nodes,
826            has_closure: &has_closure,
827            round,
828            attempted: &attempted,
829            failed_nodes: &failed_nodes,
830        };
831        let plan = strategy.next_round(&state, &net);
832        if plan.assignments.is_empty() {
833            break;
834        }
835
836        // Snapshot has_closure BEFORE dispatching, for the trace sink.
837        let has_closure_before = if trace.is_some() {
838            has_closure.clone()
839        } else {
840            HashSet::new()
841        };
842
843        if let Some(sink) = events {
844            sink.emit(&CascadeEvent::PlanComputed {
845                round,
846                assignments: plan.assignments.iter().copied().map(Edge::from).collect(),
847            });
848            for (src, tgt) in &plan.assignments {
849                sink.emit(&CascadeEvent::EdgeStarted {
850                    round,
851                    src: *src,
852                    tgt: *tgt,
853                    at: SystemTime::now(),
854                });
855            }
856        }
857
858        // Mark attempted *before* dispatch so a strategy that's called
859        // again sees the in-flight edges as already-tried.
860        for edge in &plan.assignments {
861            attempted.insert(*edge);
862        }
863
864        let outcomes = executor.dispatch(&nodes, &plan.assignments, &net);
865
866        if let Some(sink) = events {
867            for (src, tgt) in &plan.assignments {
868                match outcomes.get(&(*src, *tgt)) {
869                    Some(Ok(d)) => sink.emit(&CascadeEvent::EdgeCompleted {
870                        round,
871                        src: *src,
872                        tgt: *tgt,
873                        duration: *d,
874                    }),
875                    Some(Err(e)) => sink.emit(&CascadeEvent::EdgeFailed {
876                        round,
877                        src: *src,
878                        tgt: *tgt,
879                        error: e.clone(),
880                    }),
881                    None => sink.emit(&CascadeEvent::EdgeFailed {
882                        round,
883                        src: *src,
884                        tgt: *tgt,
885                        error: CascadeError::Copy {
886                            node: *tgt,
887                            stderr: format!("executor returned no result for edge {src} -> {tgt}"),
888                        },
889                    }),
890                }
891            }
892        }
893
894        // Round wall-time = max edge duration.
895        let round_wall = outcomes
896            .values()
897            .filter_map(|r| r.as_ref().ok())
898            .copied()
899            .max()
900            .unwrap_or(Duration::ZERO);
901        round_durations.push(round_wall);
902
903        // Apply outcomes to the cascade graph.
904        for (src, tgt) in &plan.assignments {
905            match outcomes.get(&(*src, *tgt)) {
906                Some(Ok(_)) => {
907                    has_closure.insert(*tgt);
908                    if let Some(node) = nodes.iter_mut().find(|n| n.id == *tgt) {
909                        node.parent = Some(*src);
910                    }
911                    if let Some(srcnode) = nodes.iter_mut().find(|n| n.id == *src) {
912                        if !srcnode.children.contains(tgt) {
913                            srcnode.children.push(*tgt);
914                        }
915                    }
916                }
917                Some(Err(err)) => {
918                    // Record parent linkage even on failure so error
919                    // bubbling has a path. The src is the parent that
920                    // tried to serve this target.
921                    if let Some(node) = nodes.iter_mut().find(|n| n.id == *tgt) {
922                        node.parent = Some(*src);
923                    }
924                    // Permanently mark target as failed ONLY for non-
925                    // transient errors. CascadeError::Copy = transient
926                    // (retry from another source via the strategy's
927                    // fallback). Activation / SshHandshake / Partitioned
928                    // = permanent (target is dead/unreachable; orphan
929                    // re-routing kicks in for its descendants).
930                    if !err.is_transient() {
931                        failed_nodes.insert(*tgt);
932                    }
933                    pending_errors.entry(*src).or_default().push(err.clone());
934                }
935                None => {
936                    // Executor dropped the edge — treat the same as a
937                    // transient Copy failure (let the strategy retry).
938                    let err = CascadeError::Copy {
939                        node: *tgt,
940                        stderr: format!("executor returned no result for edge {src} -> {tgt}"),
941                    };
942                    pending_errors.entry(*src).or_default().push(err);
943                }
944            }
945        }
946
947        // Bubble pending errors one level up. Each parent that has
948        // collected children-errors this round folds them into a
949        // SubtreeAggregate and the aggregate moves to its *grandparent's*
950        // bucket for next round (or to root_errors if no grandparent).
951        let mut next_pending: HashMap<NodeId, Vec<CascadeError>> = HashMap::new();
952        for (parent, errs) in pending_errors.drain() {
953            let aggregate = if errs.len() == 1 {
954                errs.into_iter().next().unwrap()
955            } else {
956                CascadeError::merge(parent, errs)
957            };
958            let grandparent = nodes.iter().find(|n| n.id == parent).and_then(|n| n.parent);
959            match grandparent {
960                Some(gp) => next_pending.entry(gp).or_default().push(aggregate),
961                None => root_errors.push(aggregate),
962            }
963        }
964        pending_errors = next_pending;
965
966        // Emit trace snapshot for this round (if a sink is wired in).
967        if let Some(sink) = trace {
968            let parent_chain: HashMap<NodeId, NodeId> = nodes
969                .iter()
970                .filter_map(|n| n.parent.map(|p| (n.id, p)))
971                .collect();
972            let snapshot = RoundSnapshot {
973                round,
974                has_closure_before,
975                plan: plan.clone(),
976                outcomes: outcomes.clone(),
977                has_closure_after: has_closure.clone(),
978                parent_chain,
979                round_duration: round_wall,
980            };
981            sink.record(&snapshot);
982        }
983
984        if let Some(sink) = events {
985            let mut has_closure_vec: Vec<NodeId> = has_closure.iter().copied().collect();
986            has_closure_vec.sort();
987            sink.emit(&CascadeEvent::RoundCompleted {
988                round,
989                duration: round_wall,
990                has_closure: has_closure_vec,
991            });
992        }
993
994        round += 1;
995
996        // Halt early if everyone converged.
997        let all_converged = nodes.iter().all(|n| has_closure.contains(&n.id));
998        if all_converged {
999            break;
1000        }
1001    }
1002
1003    // Drain any still-pending errors all the way to root.
1004    while !pending_errors.is_empty() {
1005        let mut next_pending: HashMap<NodeId, Vec<CascadeError>> = HashMap::new();
1006        for (parent, errs) in pending_errors.drain() {
1007            let aggregate = if errs.len() == 1 {
1008                errs.into_iter().next().unwrap()
1009            } else {
1010                CascadeError::merge(parent, errs)
1011            };
1012            let grandparent = nodes.iter().find(|n| n.id == parent).and_then(|n| n.parent);
1013            match grandparent {
1014                Some(gp) => next_pending.entry(gp).or_default().push(aggregate),
1015                None => root_errors.push(aggregate),
1016            }
1017        }
1018        pending_errors = next_pending;
1019    }
1020
1021    let converged: Vec<NodeId> = nodes
1022        .iter()
1023        .filter(|n| has_closure.contains(&n.id))
1024        .map(|n| n.id)
1025        .collect();
1026
1027    let failed = match root_errors.len() {
1028        0 => None,
1029        1 => Some(root_errors.into_iter().next().unwrap()),
1030        _ => Some(CascadeError::SubtreeAggregate {
1031            // synthetic root: NodeId(u32::MAX) signals "user / coordinator,
1032            // no real node corresponds." Strategies and tests should not
1033            // dereference this.
1034            node: NodeId(u32::MAX),
1035            errors: root_errors,
1036        }),
1037    };
1038
1039    if let Some(sink) = events {
1040        let failed_count = nodes
1041            .iter()
1042            .filter(|n| !has_closure.contains(&n.id))
1043            .count();
1044        sink.emit(&CascadeEvent::Finished {
1045            converged: converged.len(),
1046            failed: failed_count,
1047            rounds: round,
1048        });
1049    }
1050
1051    CascadeResult {
1052        converged,
1053        failed,
1054        rounds: round,
1055        round_durations,
1056    }
1057}

1063/// Pairs sources to targets in id-order. Ignores `NetworkProfile`.
1064/// Each source is used at most once per round, so convergence on
1065/// uniform topology is ⌈log₂(N - seeded)⌉ rounds.
1066pub struct Log2FanOut;
1067
1068impl Log2FanOut {
1069    /// ⌈log₂(N - seeded)⌉, the struct's documented uniform-topology rule:
1070    /// each source serves at most one target per round, so the informed
1071    /// set at most doubles per round. Computed in integer arithmetic
1072    /// (bit-width of `pending - 1`, no floats). Assumes `seeded >= 1`
1073    /// (the coordinator always pushes from at least one holder); with
1074    /// zero sources the cascade halts unconverged and no count applies.
1075    pub fn expected_rounds(&self, n_nodes: usize, seeded: usize) -> u32 {
1076        let pending = n_nodes.saturating_sub(seeded);
1077        if pending == 0 {
1078            return 0;
1079        }
1080        usize::BITS - (pending - 1).leading_zeros()
1081    }
1082}
1083
1084/// Parent-hop distance from the nearest seeded root, for every node.
1085/// Seeded nodes have no parent → depth 0; everyone else is one deeper
1086/// than their parent. Chains terminate on the first node with a known
1087/// depth or on a parentless (seeded) node; parents recorded on failed
1088/// deliveries are never consulted for depths of closure-holding
1089/// sources, whose chains are real relay paths.
1090fn node_depths(nodes: &[CascadeNode]) -> HashMap<NodeId, u32> {
1091    let parent_of: HashMap<NodeId, NodeId> = nodes
1092        .iter()
1093        .filter_map(|n| n.parent.map(|p| (n.id, p)))
1094        .collect();
1095    let mut depths: HashMap<NodeId, u32> = HashMap::new();
1096    for node in nodes {
1097        let mut chain = Vec::new();
1098        let mut cur = node.id;
1099        let base = loop {
1100            if let Some(&d) = depths.get(&cur) {
1101                break d;
1102            }
1103            let Some(&parent) = parent_of.get(&cur) else {
1104                break 0;
1105            };
1106            chain.push(cur);
1107            cur = parent;
1108        };
1109        depths.insert(cur, base);
1110        for (i, id) in chain.iter().rev().enumerate() {
1111            depths.insert(*id, base + (i as u32) + 1);
1112        }
1113    }
1114    depths
1115}

1117/// Sources (closure-holding, non-failed nodes) in id order.
1118fn fanout_sources(state: &CascadeState) -> Vec<NodeId> {
1119    let mut sources: Vec<NodeId> = state
1120        .nodes
1121        .iter()
1122        .filter(|n| state.has_closure.contains(&n.id))
1123        .filter(|n| !state.failed_nodes.contains(&n.id))
1124        .map(|n| n.id)
1125        .collect();
1126    sources.sort();
1127    sources
1128}

1130/// Targets (pending, non-failed nodes) in id order.
1131fn fanout_targets(state: &CascadeState) -> Vec<NodeId> {
1132    let mut targets: Vec<NodeId> = state
1133        .nodes
1134        .iter()
1135        .filter(|n| !state.has_closure.contains(&n.id))
1136        .filter(|n| !state.failed_nodes.contains(&n.id))
1137        .map(|n| n.id)
1138        .collect();
1139    targets.sort();
1140    targets
1141}

1143/// Greedy 1:1 pairing in pool order, skipping partitioned +
1144/// already-attempted edges. Each source and each target is used at
1145/// most once.
1146fn greedy_pairing(
1147    sources: &[NodeId],
1148    targets: &[NodeId],
1149    state: &CascadeState,
1150    net: &NetworkProfile,
1151) -> Vec<(NodeId, NodeId)> {
1152    let n = sources.len().min(targets.len());
1153    let mut assignments = Vec::with_capacity(n);
1154    let mut used_targets: HashSet<NodeId> = HashSet::new();
1155
1156    let mut src_iter = sources.iter().copied();
1157    for tgt in targets.iter().copied() {
1158        if used_targets.len() >= n {
1159            break;
1160        }
1161        // find next source that isn't partitioned from tgt and hasn't already
1162        // attempted this edge
1163        let chosen = loop {
1164            let Some(src) = src_iter.next() else {
1165                break None;
1166            };
1167            if net.is_partitioned(src, tgt) {
1168                continue;
1169            }
1170            if state.attempted.contains(&(src, tgt)) {
1171                continue;
1172            }
1173            break Some(src);
1174        };
1175        if let Some(src) = chosen {
1176            assignments.push((src, tgt));
1177            used_targets.insert(tgt);
1178        }
1179    }
1180    assignments
1181}

1186    }

1189        CascadePlan {

1191            assignments: greedy_pairing(&fanout_sources(state), &fanout_targets(state), state, net),
1192        }
1193    }

1196/// The log2 fanout's pairing under an explicit traversal order — the
1197/// fanout's orthogonal dimension, not a topology of its own: it names
1198/// itself exactly [`Log2FanOut`] and is measured by the same round
1199/// rule, so it is never a
1200/// [`STRATEGIES`](crate::cascade_strategies::STRATEGIES) entry.
1201///
1202/// `Default` pairs byte-for-byte like [`Log2FanOut`]. `Bfs` re-sorts
1203/// the source pool shallowest-first (nodes nearest a seeded root serve
1204/// before their own descendants); `Dfs` deepest-first (the deepest
1205/// branch is extended before moving sideways). Both re-sorts are
1206/// stable, so depth ties keep id order.
1207///
1208/// This type — not a field on [`CascadeState`] or [`NetworkProfile`] —
1209/// is how the order reaches the pairing: those structs are
1210/// exhaustively constructible downstream, so a new field on either is
1211/// a semver-major break, while a new strategy type over the unchanged
1212/// [`CascadeStrategy`] trait is purely additive.
1213pub struct OrderedFanout {
1214    order: FanoutOrder,
1215}
1216
1217impl OrderedFanout {
1218    pub fn new(order: FanoutOrder) -> Self {
1219        Self { order }
1220    }
1221
1222    /// The fanout round rule — the same ⌈log₂(N - seeded)⌉ as
1223    /// [`Log2FanOut::expected_rounds`]: the order changes which source
1224    /// serves which target within a round, never how many rounds a
1225    /// uniform fleet needs.
1226    pub fn expected_rounds(&self, n_nodes: usize, seeded: usize) -> u32 {
1227        Log2FanOut.expected_rounds(n_nodes, seeded)
1228    }
1229}
1230
1231impl CascadeStrategy for OrderedFanout {
1232    fn name(&self) -> &'static str {
1233        Log2FanOut.name()
1234    }
1235
1236    fn next_round(&self, state: &CascadeState, net: &NetworkProfile) -> CascadePlan {
1237        let mut sources = fanout_sources(state);
1238        let depths = node_depths(state.nodes);
1239        match self.order {
1240            FanoutOrder::Bfs => sources.sort_by_key(|id| depths[id]),
1241            FanoutOrder::Dfs => sources.sort_by_key(|id| std::cmp::Reverse(depths[id])),
1242            FanoutOrder::Default => {}
1243        }
1244        let targets = fanout_targets(state);
1245        CascadePlan {
1246            round: state.round,
1247            assignments: greedy_pairing(&sources, &targets, state, net),
1248        }
1249    }
1250}

1526    /// Depth of every served node by walking parent chains to a
1527    /// seeded root.
1528    fn relay_depth(parent_chain: &HashMap<NodeId, NodeId>) -> u32 {
1529        parent_chain
1530            .keys()
1531            .map(|start| {
1532                let mut depth = 0u32;
1533                let mut cur = *start;
1534                while let Some(&parent) = parent_chain.get(&cur) {
1535                    depth += 1;
1536                    cur = parent;
1537                }
1538                depth
1539            })
1540            .max()
1541            .unwrap_or(0)
1542    }
