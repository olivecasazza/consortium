1//! cast — NixOS / nix-darwin deployment orchestration CLI.
2//!
3//! Usage:
4//!   cast --flake . deploy --on @darwin switch   # build, copy, activate a group
5//!   cast deploy --on hp[01-03] switch           # bracket notation
6//!   cast build --on contra                      # build only
7//!   cast eval --on mm[01-05]                    # show what would be built
8//!   cast health                                 # probe all builders
9//!   cast status --on hp[01-03]                  # show current system versions
10//!
11//! Fleet source, in order of precedence: `--config FILE`, `./fleet.json`
12//! when it exists, the flake's `fleet` output (an `mkFleet` result), and
13//! finally nodes derived from the flake's `darwinConfigurations` /
14//! `nixosConfigurations` names. `--on` accepts `@group` references
15//! resolved through ClusterShell `groups.conf` / `groups.d`. Before a
16//! deploy, every remote target's ssh endpoint is resolved live (see
17//! [`consortium_nix::endpoint`]); a target that is this machine is
18//! activated locally without copy or ssh.
19

175pub fn run() {
176    let args = Args::parse();
177    let out = CliOutput::from_args(&args.output);
178    let exec = Arc::new(ProcessExecutor::new());
179
180    let request = FleetRequest {
181        config: args.config.as_deref(),
182        flake: args.flake.as_deref(),
183        user: args.user.as_deref(),
184        cwd: Path::new("."),
185    };
186    let (config, source) = match load_fleet(&request, &*exec) {
187        Ok(loaded) => loaded,
188        Err(e) => {
189            out.error(format!("{}", e));
190            eprintln!(
191                "hint: pass --config FILE, or run inside a flake (or pass --flake REF) \
192                 that has a `fleet` output or darwinConfigurations / nixosConfigurations"
193            );
194            process::exit(1);
195        }
196    };
197    if args.output.verbose > 0 {
198        eprintln!("fleet: {} ({} node(s))", source, config.nodes.len());
199    }
200
201    let nix_args = nix_args_from_flags(&args.nix_args, &args.darwin_nix_args, &args.nixos_nix_args);
202
203    let result = match args.command {
204        Commands::Eval { on, tag } => cmd_eval(&config, on.as_deref(), &tag),
205        Commands::Build {
206            on,
207            tag,
208            builders,
209            fanout,
210        } => cmd_deploy(
211            exec,
212            &config,
213            on.as_deref(),
214            &tag,
215            "build",
216            builders,
217            fanout,
218            false,
219            2,
220            None,
221            nix_args,
222        ),
223        Commands::Deploy {
224            on,
225            tag,
226            action,
227            builders,
228            fanout,
229            cascade,
230            cascade_fanout,
231            activate_timeout,
232        } => cmd_deploy(
233            exec,
234            &config,
235            on.as_deref(),
236            &tag,
237            &action,
238            builders,
239            fanout,
240            cascade,
241            cascade_fanout,
242            activate_timeout.map(Duration::from_secs),
243            nix_args,
244        ),
245        Commands::Health => cmd_health(&config),
246        Commands::Status { on, tag } => cmd_status(&config, on.as_deref(), &tag),
247    };
248
249    if let Err(e) = result {
250        eprintln!("error: {}", e);
251        process::exit(1);
252    }
253}

255/// Inputs deciding where the fleet comes from.
256struct FleetRequest<'a> {
257    /// `--config FILE`.
258    config: Option<&'a Path>,
259    /// `--flake REF`.
260    flake: Option<&'a str>,
261    /// `--user USER`.
262    user: Option<&'a str>,
263    /// Directory holding the implicit `fleet.json`.
264    cwd: &'a Path,
265}
266
267/// Load the fleet by precedence: `--config`, `<cwd>/fleet.json`, then
268/// discovery from the flake (`fleet` output, else configuration names).
269/// Applies the `--flake` and `--user` overrides. Returns the config and a
270/// description of its source.
271fn load_fleet(
272    req: &FleetRequest<'_>,
273    exec: &dyn Executor,
274) -> anyhow::Result<(FleetConfig, String)> {
275    let default_user = req
276        .user
277        .map(str::to_string)
278        .or_else(|| std::env::var("USER").ok())
279        .unwrap_or_else(|| "root".to_string());
280
281    let (mut config, source) = if let Some(path) = req.config {
282        (FleetConfig::from_file(path)?, path.display().to_string())
283    } else {
284        let implicit = req.cwd.join("fleet.json");
285        if implicit.is_file() {
286            (
287                FleetConfig::from_file(&implicit)?,
288                implicit.display().to_string(),
289            )
290        } else {
291            let flake = req.flake.unwrap_or(".");
292            let (config, origin) = fleet_source::discover_fleet(exec, flake, &default_user)?;
293            (config, format!("{} of {}", origin, flake))
294        }
295    };
296
297    if let Some(flake) = req.flake {
298        config.flake_uri = flake.to_string();
299    }
300    if let Some(user) = req.user {
301        for node in config.nodes.values_mut() {
302            node.target_user = user.to_string();
303        }
304    }
305    Ok((config, source))
306}

308/// Whitespace-split the repeatable `--*-nix-args` flags into [`NixArgs`];
309/// `--nix-args` words precede the platform-specific ones.
310fn nix_args_from_flags(all: &[String], darwin: &[String], nixos: &[String]) -> NixArgs {
311    let words = |groups: &[&[String]]| -> Vec<String> {
312        groups
313            .iter()
314            .flat_map(|g| g.iter())
315            .flat_map(|s| s.split_whitespace())
316            .map(str::to_string)
317            .collect()
318    };
319    NixArgs {
320        darwin: words(&[all, darwin]),
321        nixos: words(&[all, nixos]),
322    }
323}

325/// Resolve target nodes from --on and --tag flags, falling back to all nodes.
326///
327/// `--on` tokens may be `@group` references; expanded names not found in
328/// the fleet fall back to their bare form (`host.local` → `host`).
329fn resolve_targets(
330    config: &FleetConfig,
331    on: Option<&str>,
332    tags: &[String],
333) -> anyhow::Result<Vec<String>> {
334    let mut targets = Vec::new();
335
336    if let Some(pattern) = on {
337        let ns = groups::expand_pattern(pattern, &groups::group_nodes)?;
338        for name in ns.iter() {
339            let bare = endpoint::bare_name(&name).to_string();
340            let resolved = if config.nodes.contains_key(&name) {
341                name
342            } else if config.nodes.contains_key(&bare) {
343                bare
344            } else {
345                name
346            };
347            if !targets.contains(&resolved) {
348                targets.push(resolved);
349            }
350        }
351    }
352
353    if !tags.is_empty() {
354        let tag_nodes = config.nodes_by_tags(tags);
355        for node in tag_nodes {
356            if !targets.contains(&node.name) {
357                targets.push(node.name.clone());
358            }
359        }
360    }
361
362    // Default to all nodes if nothing specified
363    if targets.is_empty() {
364        targets = config.node_names();
365    }
366
367    // Validate all targets exist in config
368    for name in &targets {
369        if !config.nodes.contains_key(name) {
370            anyhow::bail!(
371                "unknown node '{}' (available: {})",
372                name,
373                config.node_names().join(", ")
374            );
375        }
376    }
377
378    Ok(targets)
379}

381fn cmd_eval(config: &FleetConfig, on: Option<&str>, tags: &[String]) -> anyhow::Result<()> {
382    let targets = resolve_targets(config, on, tags)?;
383
384    println!("Evaluating {} host(s):", targets.len());
385    for name in &targets {
386        let node = &config.nodes[name];
387        println!(
388            "  {} → {} ({}@{}, tags: [{}])",
389            name,
390            node.system,
391            node.target_user,
392            node.target_host,
393            node.tags.join(", ")
394        );
395        if let Some(drv) = &node.drv_path {
396            println!("    drv: {}", drv);
397        }
398    }
399
400    Ok(())
401}

403/// Live endpoint resolution for a deploy: hosts that are this machine are
404/// activated locally; every other host's `target_host` is replaced by the
405/// first endpoint accepting ssh (candidates extended through `lookup`).
406/// Returns the local hosts and the hosts that could not be reached (with
407/// the error to report).
408fn resolve_endpoints(
409    exec: &dyn Executor,
410    lookup: endpoint::Lookup<'_>,
411    config: &mut FleetConfig,
412    targets: &[String],
413) -> (Vec<String>, Vec<(String, String)>) {
414    let local_hostname = endpoint::local_hostname(exec);
415    let mut local_hosts = Vec::new();
416    let mut remote: Vec<String> = Vec::new();
417    for name in targets {
418        match &local_hostname {
419            Some(local) if endpoint::is_local_node(name, local) => local_hosts.push(name.clone()),
420            _ => remote.push(name.clone()),
421        }
422    }
423
424    let resolver = EndpointResolver::new(exec, lookup);
425    let results: Vec<(String, consortium_nix::Result<String>)> = std::thread::scope(|scope| {
426        let handles: Vec<_> = remote
427            .iter()
428            .map(|name| {
429                let node = &config.nodes[name];
430                let resolver = &resolver;
431                scope.spawn(move || (name.clone(), resolver.resolve(node)))
432            })
433            .collect();
434        handles
435            .into_iter()
436            .map(|h| h.join().expect("endpoint resolution thread panicked"))
437            .collect()
438    });
439
440    let mut unreachable = Vec::new();
441    for (name, result) in results {
442        match result {
443            Ok(host) => {
444                config
445                    .nodes
446                    .get_mut(&name)
447                    .expect("validated target")
448                    .target_host = host
449            }
450            Err(e) => unreachable.push((name, e.to_string())),
451        }
452    }
453    (local_hosts, unreachable)
454}

456#[allow(clippy::too_many_arguments)]
457fn cmd_deploy(
458    exec: Arc<ProcessExecutor>,
459    config: &FleetConfig,
460    on: Option<&str>,
461    tags: &[String],
462    action_str: &str,
463    use_builders: bool,
464    fanout: usize,
465    cascade: bool,
466    cascade_fanout: u32,
467    activate_timeout: Option<Duration>,
468    nix_args: NixArgs,
469) -> anyhow::Result<()> {
470    let mut targets = resolve_targets(config, on, tags)?;
471    let action: DeployAction = action_str.parse().map_err(|_| {
472        anyhow::anyhow!(
473            "invalid action '{}' (try: switch, boot, test, dry-activate, build)",
474            action_str
475        )
476    })?;
477
478    let mut config = config.clone();
479    let mut options = DeployOptions::new().nix_args(nix_args);
480    if let Some(limit) = activate_timeout {
481        options = options.activate_timeout(limit);
482    }
483    let mut unreachable = Vec::new();
484    if action != DeployAction::Build {
485        println!("Resolving ssh endpoints for {} host(s)...", targets.len());
486        let (local_hosts, failed) =
487            resolve_endpoints(&*exec, &endpoint::system_lookup, &mut config, &targets);
488        unreachable = failed;
489        targets.retain(|t| !unreachable.iter().any(|(u, _)| u == t));
490        options = options.local_hosts(local_hosts);
491    }
492
493    println!(
494        "Deploying {} host(s) with action '{}'{}:",
495        targets.len(),
496        action,
497        if cascade {
498            format!(
499                " [cascade copy fanout={} — peer-to-peer fan-out]",
500                cascade_fanout
501            )
502        } else {
503            String::new()
504        }
505    );
506    for name in &targets {
507        let node = &config.nodes[name];
508        if options.is_local(name) {
509            println!("  {} → local (this machine)", name);
510        } else if action == DeployAction::Build {
511            println!("  {}", name);
512        } else {
513            println!("  {} → {}@{}", name, node.target_user, node.target_host);
514        }
515    }
516    for (name, err) in &unreachable {
517        eprintln!("  skipping {}: {}", name, err);
518    }
519
520    let report = if targets.is_empty() {
521        None
522    } else if cascade && action != DeployAction::Build {
523        // Determine seed addr — the host running cast IS the seed
524        // (closure was built locally, or fetched here). The display
525        // name is just for the live UI; NixCopyExecutor uses local
526        // `nix copy` regardless.
527        let seed_addr = std::env::var("USER")
528            .map(|u| format!("{}@localhost", u))
529            .unwrap_or_else(|_| "localhost".into());
530        Some(consortium_nix::deploy_with_cascade_options(
531            exec,
532            &config,
533            &targets,
534            action,
535            fanout,
536            use_builders,
537            cascade_fanout,
538            &seed_addr,
539            None, // event sink — deferred until LiveTreeRenderer wiring
540            &options,
541        )?)
542    } else {
543        Some(consortium_nix::deploy_with_options(
544            exec,
545            &config,
546            &targets,
547            action,
548            fanout,
549            use_builders,
550            &options,
551        )?)
552    };
553
554    println!();
555    let mut failed = unreachable.len();
556    if let Some(report) = report {
557        failed += report.failure_count();
558        if report.is_success() {
559            println!("Deployment successful!");
560            println!("  Built: {}", report.built.len());
561            if action != DeployAction::Build {
562                println!("  Copied: {}", report.copied.len());
563                println!("  Activated: {}", report.activated.len());
564            }
565        } else {
566            println!(
567                "Deployment completed with {} failure(s):",
568                report.failure_count()
569            );
570            for (name, err) in &report.eval_failures {
571                eprintln!("  eval failed on {}: {}", name, err);
572            }
573            for (name, err) in &report.build_failures {
574                eprintln!("  build failed on {}: {}", name, err);
575            }
576            for (name, err) in &report.copy_failures {
577                eprintln!("  copy failed to {}: {}", name, err);
578            }
579            for (name, err) in &report.activation_failures {
580                eprintln!("  activation failed on {}: {}", name, err);
581            }
582        }
583    }
584    if !unreachable.is_empty() {
585        println!("Unreachable ({} host(s)):", unreachable.len());
586        for (name, err) in &unreachable {
587            eprintln!("  {}: {}", name, err);
588        }
589    }
590    if failed > 0 {
591        process::exit(1);
592    }
593
594    Ok(())
595}

597fn cmd_health(config: &FleetConfig) -> anyhow::Result<()> {
598    if config.builders.is_empty() {
599        println!("No builders configured.");
600        return Ok(());
601    }
602
603    println!("Probing {} builder(s):", config.builders.len());
604
605    let statuses = health::check_builders(config);
606    let healthy_count = statuses.iter().filter(|s| s.healthy).count();
607
608    for status in &statuses {
609        let icon = if status.healthy { "ok" } else { "FAIL" };
610        let latency = status
611            .latency_ms
612            .map(|ms| format!(" ({}ms)", ms))
613            .unwrap_or_default();
614        let error = status
615            .error
616            .as_ref()
617            .map(|e| format!(" - {}", e))
618            .unwrap_or_default();
619
620        println!(
621            "  [{}] {}@{}{}{} (jobs:{}, speed:{}x)",
622            icon,
623            status.builder.user,
624            status.builder.host,
625            latency,
626            error,
627            status.builder.max_jobs,
628            status.builder.speed_factor,
629        );
630    }
631
632    println!("\n{}/{} builders healthy", healthy_count, statuses.len());
633
634    Ok(())
635}

637fn cmd_status(config: &FleetConfig, on: Option<&str>, tags: &[String]) -> anyhow::Result<()> {
638    let targets = resolve_targets(config, on, tags)?;
639
640    println!("Querying {} host(s):", targets.len());
641
642    for name in &targets {
643        let node = &config.nodes[name];
644        match consortium_nix::eval::query_current_system(
645            &ProcessExecutor::new(),
646            &node.target_host,
647            &node.target_user,
648        ) {
649            Ok(Some(path)) => println!("  {} → {}", name, path),
650            Ok(None) => println!("  {} → (unknown)", name),
651            Err(e) => println!("  {} → error: {}", name, e),
652        }
653    }
654
655    Ok(())
656}

806    #[test]
807    fn resolve_endpoints_splits_local_and_rewrites_remote_target_host() {
808        let mut config = FleetConfig::from_json(FLEET_JSON).unwrap();
809        // Add a node whose name is this machine.
810        let mut me = config.nodes["filehost"].clone();
811        me.name = "Workstation".into();
812        me.target_host = "Workstation".into();
813        config.nodes.insert("Workstation".into(), me);
814        let mut dead = config.nodes["filehost"].clone();
815        dead.name = "dead".into();
816        dead.target_host = "198.51.100.7".into();
817        config.nodes.insert("dead".into(), dead);
818
819        let exec = ScriptedExecutor::new()
820            .on("hostname", ExecOutput::ok("workstation.lan\n"))
821            .on("198.51.100.7", ExecOutput::new(255, "", "refused"))
822            .on("dead.local", ExecOutput::new(255, "", "refused"))
823            .on(" dead 'true'", ExecOutput::new(255, "", "refused"))
824            .on("192.0.2.1", ExecOutput::new(255, "", "stale"))
825            .on("filehost.local", ExecOutput::ok(""));
826        let targets = vec![
827            "filehost".to_string(),
828            "Workstation".to_string(),
829            "dead".to_string(),
830        ];
831        let (local, unreachable) =
832            resolve_endpoints(&exec, &|_: &str| vec![], &mut config, &targets);
833
834        assert_eq!(local, vec!["Workstation"]);
835        assert_eq!(config.nodes["filehost"].target_host, "filehost.local");
836        assert_eq!(unreachable.len(), 1);
837        assert_eq!(unreachable[0].0, "dead");
838        assert!(
839            unreachable[0]
840                .1
841                .contains("tried: 198.51.100.7, dead.local, dead"),
842            "{}",
843            unreachable[0].1
844        );
845        // The local host was never probed over ssh.
846        exec.assert_not_invoked_containing("Workstation");
847    }
