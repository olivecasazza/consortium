1//! cast — NixOS / nix-darwin deployment orchestration CLI.
2//!
3//! Usage:
4//!   cast --flake . deploy --on @darwin switch   # build, copy, activate a group
5//!   cast deploy --on hp[01-03] switch           # bracket notation
6//!   cast build --on contra                      # build only
7//!   cast eval --on mm[01-05]                    # show what would be built
8//!   cast health                                 # probe all builders
9//!   cast status --on hp[01-03]                  # show current system versions
10//!
11//! Fleet source, in order of precedence: `--config FILE`, `./fleet.json`
12//! when it exists, the flake's `fleet` output (an `mkFleet` result), and
13//! finally nodes derived from the flake's `darwinConfigurations` /
14//! `nixosConfigurations` names. `--on` accepts `@group` references
15//! resolved through ClusterShell `groups.conf` / `groups.d`. Before a
16//! deploy, every remote target's ssh endpoint is resolved live (see
17//! [`consortium_nix::endpoint`]); a target that is this machine is
18//! activated locally without copy or ssh.
19

182pub fn run() {
183    let args = Args::parse();
184    let out = CliOutput::from_args(&args.output);
185    let exec = Arc::new(ProcessExecutor::new());
186
187    let result = match args.command {
188        // Cascade streams are self-contained: replay / verify / live never
189        // touch the fleet, so this arm runs before fleet discovery.
190        Commands::Cascade { command } => cascade::dispatch(command),
191        command => {
192            let request = FleetRequest {
193                config: args.config.as_deref(),
194                flake: args.flake.as_deref(),
195                user: args.user.as_deref(),
196                cwd: Path::new("."),
197            };
198            let (config, source) = match load_fleet(&request, &*exec) {
199                Ok(loaded) => loaded,
200                Err(e) => {
201                    out.error(format!("{}", e));
202                    eprintln!(
203                        "hint: pass --config FILE, or run inside a flake (or pass --flake REF) \
204                         that has a `fleet` output or darwinConfigurations / nixosConfigurations"
205                    );
206                    process::exit(1);
207                }
208            };
209            if args.output.verbose > 0 {
210                eprintln!("fleet: {} ({} node(s))", source, config.nodes.len());
211            }
212
213            let nix_args =
214                nix_args_from_flags(&args.nix_args, &args.darwin_nix_args, &args.nixos_nix_args);
215
216            match command {
217                Commands::Eval { on, tag } => cmd_eval(&config, on.as_deref(), &tag),
218                Commands::Build {
219                    on,
220                    tag,
221                    builders,
222                    fanout,
223                } => cmd_deploy(
224                    exec,
225                    &config,
226                    on.as_deref(),
227                    &tag,
228                    "build",
229                    builders,
230                    fanout,
231                    false,
232                    2,
233                    None,
234                    nix_args,
235                ),
236                Commands::Deploy {
237                    on,
238                    tag,
239                    action,
240                    builders,
241                    fanout,
242                    cascade,
243                    cascade_fanout,
244                    activate_timeout,
245                } => cmd_deploy(
246                    exec,
247                    &config,
248                    on.as_deref(),
249                    &tag,
250                    &action,
251                    builders,
252                    fanout,
253                    cascade,
254                    cascade_fanout,
255                    activate_timeout.map(Duration::from_secs),
256                    nix_args,
257                ),
258                Commands::Health => cmd_health(&config),
259                Commands::Status { on, tag } => cmd_status(&config, on.as_deref(), &tag),
260                // Handled before fleet discovery; unreachable here.
261                Commands::Cascade { .. } => unreachable!(),
262            }
263        }
264    };
265
266    if let Err(e) = result {
267        eprintln!("error: {}", e);
268        process::exit(1);
269    }
270}

272/// Inputs deciding where the fleet comes from.
273struct FleetRequest<'a> {
274    /// `--config FILE`.
275    config: Option<&'a Path>,
276    /// `--flake REF`.
277    flake: Option<&'a str>,
278    /// `--user USER`.
279    user: Option<&'a str>,
280    /// Directory holding the implicit `fleet.json`.
281    cwd: &'a Path,
282}
283
284/// Load the fleet by precedence: `--config`, `<cwd>/fleet.json`, then
285/// discovery from the flake (`fleet` output, else configuration names).
286/// Applies the `--flake` and `--user` overrides. Returns the config and a
287/// description of its source.
288fn load_fleet(
289    req: &FleetRequest<'_>,
290    exec: &dyn Executor,
291) -> anyhow::Result<(FleetConfig, String)> {
292    let default_user = req
293        .user
294        .map(str::to_string)
295        .or_else(|| std::env::var("USER").ok())
296        .unwrap_or_else(|| "root".to_string());
297
298    let (mut config, source) = if let Some(path) = req.config {
299        (FleetConfig::from_file(path)?, path.display().to_string())
300    } else {
301        let implicit = req.cwd.join("fleet.json");
302        if implicit.is_file() {
303            (
304                FleetConfig::from_file(&implicit)?,
305                implicit.display().to_string(),
306            )
307        } else {
308            let flake = req.flake.unwrap_or(".");
309            let (config, origin) = fleet_source::discover_fleet(exec, flake, &default_user)?;
310            (config, format!("{} of {}", origin, flake))
311        }
312    };
313
314    if let Some(flake) = req.flake {
315        config.flake_uri = flake.to_string();
316    }
317    if let Some(user) = req.user {
318        for node in config.nodes.values_mut() {
319            node.target_user = user.to_string();
320        }
321    }
322    Ok((config, source))
323}

325/// Whitespace-split the repeatable `--*-nix-args` flags into [`NixArgs`];
326/// `--nix-args` words precede the platform-specific ones.
327fn nix_args_from_flags(all: &[String], darwin: &[String], nixos: &[String]) -> NixArgs {
328    let words = |groups: &[&[String]]| -> Vec<String> {
329        groups
330            .iter()
331            .flat_map(|g| g.iter())
332            .flat_map(|s| s.split_whitespace())
333            .map(str::to_string)
334            .collect()
335    };
336    NixArgs {
337        darwin: words(&[all, darwin]),
338        nixos: words(&[all, nixos]),
339    }
340}

342/// Resolve target nodes from --on and --tag flags, falling back to all nodes.
343///
344/// `--on` tokens may be `@group` references; expanded names not found in
345/// the fleet fall back to their bare form (`host.local` → `host`).
346fn resolve_targets(
347    config: &FleetConfig,
348    on: Option<&str>,
349    tags: &[String],
350) -> anyhow::Result<Vec<String>> {
351    let mut targets = Vec::new();
352
353    if let Some(pattern) = on {
354        let ns = groups::expand_pattern(pattern, &groups::group_nodes)?;
355        for name in ns.iter() {
356            let bare = endpoint::bare_name(&name).to_string();
357            let resolved = if config.nodes.contains_key(&name) {
358                name
359            } else if config.nodes.contains_key(&bare) {
360                bare
361            } else {
362                name
363            };
364            if !targets.contains(&resolved) {
365                targets.push(resolved);
366            }
367        }
368    }
369
370    if !tags.is_empty() {
371        let tag_nodes = config.nodes_by_tags(tags);
372        for node in tag_nodes {
373            if !targets.contains(&node.name) {
374                targets.push(node.name.clone());
375            }
376        }
377    }
378
379    // Default to all nodes if nothing specified
380    if targets.is_empty() {
381        targets = config.node_names();
382    }
383
384    // Validate all targets exist in config
385    for name in &targets {
386        if !config.nodes.contains_key(name) {
387            anyhow::bail!(
388                "unknown node '{}' (available: {})",
389                name,
390                config.node_names().join(", ")
391            );
392        }
393    }
394
395    Ok(targets)
396}

398fn cmd_eval(config: &FleetConfig, on: Option<&str>, tags: &[String]) -> anyhow::Result<()> {
399    let targets = resolve_targets(config, on, tags)?;
400
401    println!("Evaluating {} host(s):", targets.len());
402    for name in &targets {
403        let node = &config.nodes[name];
404        println!(
405            "  {} → {} ({}@{}, tags: [{}])",
406            name,
407            node.system,
408            node.target_user,
409            node.target_host,
410            node.tags.join(", ")
411        );
412        if let Some(drv) = &node.drv_path {
413            println!("    drv: {}", drv);
414        }
415    }
416
417    Ok(())

420/// Live endpoint resolution for a deploy: hosts that are this machine are
421/// activated locally; every other host's `target_host` is replaced by the
422/// first endpoint accepting ssh (candidates extended through `lookup`).
423/// Returns the local hosts and the hosts that could not be reached (with
424/// the error to report).
425fn resolve_endpoints(
426    exec: &dyn Executor,
427    lookup: endpoint::Lookup<'_>,
428    config: &mut FleetConfig,
429    targets: &[String],
430) -> (Vec<String>, Vec<(String, String)>) {
431    let local_hostname = endpoint::local_hostname(exec);
432    let mut local_hosts = Vec::new();
433    let mut remote: Vec<String> = Vec::new();
434    for name in targets {
435        match &local_hostname {
436            Some(local) if endpoint::is_local_node(name, local) => local_hosts.push(name.clone()),
437            _ => remote.push(name.clone()),
438        }
439    }
440
441    let resolver = EndpointResolver::new(exec, lookup);
442    let results: Vec<(String, consortium_nix::Result<String>)> = std::thread::scope(|scope| {
443        let handles: Vec<_> = remote
444            .iter()
445            .map(|name| {
446                let node = &config.nodes[name];
447                let resolver = &resolver;
448                scope.spawn(move || (name.clone(), resolver.resolve(node)))
449            })
450            .collect();
451        handles
452            .into_iter()
453            .map(|h| h.join().expect("endpoint resolution thread panicked"))
454            .collect()
455    });
456
457    let mut unreachable = Vec::new();
458    for (name, result) in results {
459        match result {
460            Ok(host) => {
461                config
462                    .nodes
463                    .get_mut(&name)
464                    .expect("validated target")
465                    .target_host = host
466            }
467            Err(e) => unreachable.push((name, e.to_string())),
468        }
469    }
470    (local_hosts, unreachable)
471}

473#[allow(clippy::too_many_arguments)]
474fn cmd_deploy(
475    exec: Arc<ProcessExecutor>,
476    config: &FleetConfig,
477    on: Option<&str>,
478    tags: &[String],
479    action_str: &str,
480    use_builders: bool,
481    fanout: usize,
482    cascade: bool,
483    cascade_fanout: u32,
484    activate_timeout: Option<Duration>,
485    nix_args: NixArgs,
486) -> anyhow::Result<()> {
487    let mut targets = resolve_targets(config, on, tags)?;
488    let action: DeployAction = action_str.parse().map_err(|_| {
489        anyhow::anyhow!(
490            "invalid action '{}' (try: switch, boot, test, dry-activate, build)",
491            action_str
492        )
493    })?;
494
495    let mut config = config.clone();
496    let mut options = DeployOptions::new().nix_args(nix_args);
497    if let Some(limit) = activate_timeout {
498        options = options.activate_timeout(limit);
499    }
500    let mut unreachable = Vec::new();
501    if action != DeployAction::Build {
502        println!("Resolving ssh endpoints for {} host(s)...", targets.len());
503        let (local_hosts, failed) =
504            resolve_endpoints(&*exec, &endpoint::system_lookup, &mut config, &targets);
505        unreachable = failed;
506        targets.retain(|t| !unreachable.iter().any(|(u, _)| u == t));
507        options = options.local_hosts(local_hosts);
508    }
509
510    println!(
511        "Deploying {} host(s) with action '{}'{}:",
512        targets.len(),
513        action,
514        if cascade {
515            format!(
516                " [cascade copy fanout={} — peer-to-peer fan-out]",
517                cascade_fanout
518            )
519        } else {
520            String::new()
521        }
522    );
523    for name in &targets {
524        let node = &config.nodes[name];
525        if options.is_local(name) {
526            println!("  {} → local (this machine)", name);
527        } else if action == DeployAction::Build {
528            println!("  {}", name);
529        } else {
530            println!("  {} → {}@{}", name, node.target_user, node.target_host);
531        }
532    }
533    for (name, err) in &unreachable {
534        eprintln!("  skipping {}: {}", name, err);
535    }
536
537    let report = if targets.is_empty() {
538        None
539    } else if cascade && action != DeployAction::Build {
540        // Determine seed addr — the host running cast IS the seed
541        // (closure was built locally, or fetched here). The display
542        // name is just for the live UI; NixCopyExecutor uses local
543        // `nix copy` regardless.
544        let seed_addr = std::env::var("USER")
545            .map(|u| format!("{}@localhost", u))
546            .unwrap_or_else(|_| "localhost".into());
547        Some(consortium_nix::deploy_with_cascade_options(
548            exec,
549            &config,
550            &targets,
551            action,
552            fanout,
553            use_builders,
554            cascade_fanout,
555            &seed_addr,
556            None, // event sink — deferred until LiveTreeRenderer wiring
557            &options,
558        )?)
559    } else {
560        Some(consortium_nix::deploy_with_options(
561            exec,
562            &config,
563            &targets,
564            action,
565            fanout,
566            use_builders,
567            &options,
568        )?)
569    };
570
571    println!();
572    let mut failed = unreachable.len();
573    if let Some(report) = report {
574        failed += report.failure_count();
575        if report.is_success() {
576            println!("Deployment successful!");
577            println!("  Built: {}", report.built.len());
578            if action != DeployAction::Build {
579                println!("  Copied: {}", report.copied.len());
580                println!("  Activated: {}", report.activated.len());
581            }
582        } else {
583            println!(
584                "Deployment completed with {} failure(s):",
585                report.failure_count()
586            );
587            for (name, err) in &report.eval_failures {
588                eprintln!("  eval failed on {}: {}", name, err);
589            }
590            for (name, err) in &report.build_failures {
591                eprintln!("  build failed on {}: {}", name, err);
592            }
593            for (name, err) in &report.copy_failures {
594                eprintln!("  copy failed to {}: {}", name, err);
595            }
596            for (name, err) in &report.activation_failures {
597                eprintln!("  activation failed on {}: {}", name, err);
598            }
599        }
600    }
601    if !unreachable.is_empty() {
602        println!("Unreachable ({} host(s)):", unreachable.len());
603        for (name, err) in &unreachable {
604            eprintln!("  {}: {}", name, err);
605        }
606    }
607    if failed > 0 {
608        process::exit(1);
609    }
610
611    Ok(())
612}

614fn cmd_health(config: &FleetConfig) -> anyhow::Result<()> {
615    if config.builders.is_empty() {
616        println!("No builders configured.");
617        return Ok(());
618    }
619
620    println!("Probing {} builder(s):", config.builders.len());
621
622    let statuses = health::check_builders(config);
623    let healthy_count = statuses.iter().filter(|s| s.healthy).count();
624
625    for status in &statuses {
626        let icon = if status.healthy { "ok" } else { "FAIL" };
627        let latency = status
628            .latency_ms
629            .map(|ms| format!(" ({}ms)", ms))
630            .unwrap_or_default();
631        let error = status
632            .error
633            .as_ref()
634            .map(|e| format!(" - {}", e))
635            .unwrap_or_default();
636
637        println!(
638            "  [{}] {}@{}{}{} (jobs:{}, speed:{}x)",
639            icon,
640            status.builder.user,
641            status.builder.host,
642            latency,
643            error,
644            status.builder.max_jobs,
645            status.builder.speed_factor,
646        );
647    }
648
649    println!("\n{}/{} builders healthy", healthy_count, statuses.len());
650
651    Ok(())
652}

654fn cmd_status(config: &FleetConfig, on: Option<&str>, tags: &[String]) -> anyhow::Result<()> {
655    let targets = resolve_targets(config, on, tags)?;
656
657    println!("Querying {} host(s):", targets.len());
658
659    for name in &targets {
660        let node = &config.nodes[name];
661        match consortium_nix::eval::query_current_system(
662            &ProcessExecutor::new(),
663            &node.target_host,
664            &node.target_user,
665        ) {
666            Ok(Some(path)) => println!("  {} → {}", name, path),
667            Ok(None) => println!("  {} → (unknown)", name),
668            Err(e) => println!("  {} → error: {}", name, e),
669        }
670    }
671
672    Ok(())
673}

823    #[test]
824    fn resolve_endpoints_splits_local_and_rewrites_remote_target_host() {
825        let mut config = FleetConfig::from_json(FLEET_JSON).unwrap();
826        // Add a node whose name is this machine.
827        let mut me = config.nodes["filehost"].clone();
828        me.name = "Workstation".into();
829        me.target_host = "Workstation".into();
830        config.nodes.insert("Workstation".into(), me);
831        let mut dead = config.nodes["filehost"].clone();
832        dead.name = "dead".into();
833        dead.target_host = "198.51.100.7".into();
834        config.nodes.insert("dead".into(), dead);
835
836        let exec = ScriptedExecutor::new()
837            .on("hostname", ExecOutput::ok("workstation.lan\n"))
838            .on("198.51.100.7", ExecOutput::new(255, "", "refused"))
839            .on("dead.local", ExecOutput::new(255, "", "refused"))
840            .on(" dead 'true'", ExecOutput::new(255, "", "refused"))
841            .on("192.0.2.1", ExecOutput::new(255, "", "stale"))
842            .on("filehost.local", ExecOutput::ok(""));
843        let targets = vec![
844            "filehost".to_string(),
845            "Workstation".to_string(),
846            "dead".to_string(),
847        ];
848        let (local, unreachable) =
849            resolve_endpoints(&exec, &|_: &str| vec![], &mut config, &targets);
850
851        assert_eq!(local, vec!["Workstation"]);
852        assert_eq!(config.nodes["filehost"].target_host, "filehost.local");
853        assert_eq!(unreachable.len(), 1);
854        assert_eq!(unreachable[0].0, "dead");
855        assert!(
856            unreachable[0]
857                .1
858                .contains("tried: 198.51.100.7, dead.local, dead"),
859            "{}",
860            unreachable[0].1
861        );
862        // The local host was never probed over ssh.
863        exec.assert_not_invoked_containing("Workstation");
864    }
