1//! Adapter for the shared integration grammar.
2
3use consortium_integration::fleet::FleetConfig;
4use consortium_integration::grammar::{
5    EffectKind, FailurePolicy, Grammar, Integration, IntegrationError, Parameter, ParameterKind,
6    Phase, PhaseKind, PhaseScope, PhaseTask, Plan,
7};
8
9/// Ansible playbook execution through the uniform integration contract.
10pub struct AnsibleIntegration;
11
12static GRAMMAR: Grammar = Grammar {
13    name: "ansible",
14    about: "Ansible playbook execution",
15    targets: true,
16    failure_policy: FailurePolicy::ContinueIndependent,
17    parameters: &[
18        Parameter::required("playbook", "Playbook path on the control node"),
19        Parameter::required("environment", "Nix environment attribute name"),
20        Parameter::defaulted(
21            "check",
22            "Run Ansible check mode",
23            ParameterKind::Boolean,
24            "false",
25        ),
26        Parameter::defaulted(
27            "fanout",
28            "Maximum parallel playbook runs",
29            ParameterKind::PositiveInteger,
30            "4",
31        ),
32    ],
33    phases: &[
34        Phase::new("build").behavior(PhaseKind::Build, PhaseScope::Shared, &[EffectKind::Process]),
35        Phase::new("copy").behavior(
36            PhaseKind::Copy,
37            PhaseScope::Shared,
38            &[EffectKind::Process, EffectKind::RemoteProcess],
39        ),
40        Phase::new("execute").behavior(
41            PhaseKind::Execute,
42            PhaseScope::Target,
43            &[EffectKind::Process, EffectKind::RemoteProcess],
44        ),
45    ],
46};
47
48impl Integration for AnsibleIntegration {
49    fn grammar(&self) -> &'static Grammar {
50        &GRAMMAR
51    }
52
53    fn validate_config(&self, config: &FleetConfig) -> Result<(), IntegrationError> {
54        if config.ansible_config.is_none() {
55            return Err(IntegrationError(
56                "ansible: missing fleet configuration".into(),
57            ));
58        }
59        Ok(())
60    }
61
62    fn task(
63        &self,
64        config: &FleetConfig,
65        plan: &Plan,
66        phase: &Phase,
67        resource: &str,
68    ) -> Result<PhaseTask, IntegrationError> {
69        let c = config
70            .ansible_config
71            .as_ref()
72            .ok_or_else(|| IntegrationError("ansible: missing configuration".into()))?;
73        let env = plan.value("environment");
74        Ok(match phase.name {
75            "build" => PhaseTask::new(
76                format!("build-ansible-env:{env}"),
77                crate::tasks::NixBuildAnsibleEnvTask::new(env, &config.flake_uri),
78            ),
79            "copy" => PhaseTask::new(
80                format!("copy-ansible-env:{env}"),
81                crate::tasks::NixCopyAnsibleEnvTask {
82                    env_name: env.into(),
83                    target_host: c.control_node.clone(),
84                    target_user: "root".into(),
85                },
86            ),
87            "execute" => PhaseTask::new(
88                format!("run-playbook:{resource}"),
89                crate::tasks::AnsiblePlaybookTask::new(
90                    resource,
91                    plan.value("playbook"),
92                    env,
93                    &c.control_node,
94                    "root",
95                )
96                .with_check(plan.flag("check")),
97            ),
98            _ => return Err(IntegrationError("unsupported ansible phase".into())),
99        })
100    }
101}
