119impl Integration for NixIntegration {
120    fn grammar(&self) -> &'static Grammar {
121        &GRAMMAR
122    }
123
124    fn validate_config(&self, _config: &FleetConfig) -> Result<(), IntegrationError> {
125        Ok(())
126    }
127
128    fn validate_plan(&self, config: &FleetConfig, plan: &Plan) -> Result<(), IntegrationError> {
129        for target in &plan.targets {
130            if config.nodes[target].target_port == Some(0) {
131                return Err(IntegrationError(format!("invalid SSH port for {target}")));
132            }
133        }
134        if plan.flag("cascade") {
135            if plan.value("action") == "build" || plan.value("seed").trim().is_empty() {
136                return Err(IntegrationError(
137                    "cascade requires a seed and a deployment action".into(),
138                ));
139            }
140            u32::try_from(plan.positive_integer("cascade-fanout")?)
141                .map_err(|_| IntegrationError("cascade-fanout exceeds u32".into()))?;
142        }
143        for key in ["nixos-nix-args", "darwin-nix-args"] {
144            argument_words(plan.value(key))?;
145        }
146        if let Some(local) = plan.parameters.get("local") {
147            for host in local_words(local)? {
148                if !plan.targets.iter().any(|target| target == &host) {
149                    return Err(IntegrationError(format!(
150                        "nix: local host {host:?} is not a selected target"
151                    )));
152                }
153            }
154        }
155        Ok(())
156    }
157
158    fn configure(
159        &self,
160        _config: &FleetConfig,
161        plan: &Plan,
162        context: &DagContext,
163    ) -> Result<(), IntegrationError> {
164        let action: crate::config::DeployAction = plan
165            .value("action")
166            .parse()
167            .map_err(|e: crate::config::ConfigError| IntegrationError(e.to_string()))?;
168        let mut options = crate::DeployOptions::new().nix_args(crate::NixArgs {
169            nixos: argument_words(plan.value("nixos-nix-args"))?,
170            darwin: argument_words(plan.value("darwin-nix-args"))?,
171        });
172        if let Some(local) = plan.parameters.get("local") {
173            options = options.local_hosts(local_words(local)?);
174        }
175        if plan.parameters.contains_key("activate-timeout") {
176            options = options.activate_timeout(plan.duration("activate-timeout")?);
177        }
178        context.set_state("integration_plan", plan.clone());
179        context.set_state(
180            "machines_file_lease",
181            std::sync::Arc::new(std::sync::Mutex::new(
182                None::<consortium_integration::TemporaryFile>,
183            )),
184        );
185        context.set_state("action", action);
186        context.set_state("deploy_options", options);
187        Ok(())
188    }
189    fn cleanup(&self, context: &DagContext) -> Result<(), IntegrationError> {
190        let lease: std::sync::Arc<std::sync::Mutex<Option<consortium_integration::TemporaryFile>>> =
191            context
192                .get_state("machines_file_lease")
193                .ok_or_else(|| IntegrationError("missing machines file lease".into()))?;
194        if let Some(file) = lease.lock().unwrap().take() {
195            file.close()
196                .map_err(|e| IntegrationError(format!("machines file cleanup failed: {e}")))?;
197        }
198        Ok(())
199    }
200    fn task(
201        &self,
202        _config: &FleetConfig,
203        plan: &Plan,
204        phase: &Phase,
205        resource: &str,
206    ) -> Result<PhaseTask, IntegrationError> {
207        Ok(match phase.name {
208            "builders" => PhaseTask::new(
209                "builders",
210                consortium::dag::FnTask::new("prepare distributed builders", |context| {
211                    let config: FleetConfig = context.get_state("fleet_config").unwrap();
212                    let exec: std::sync::Arc<dyn consortium_integration::Executor> =
213                        context.get_state("executor").unwrap();
214                    match crate::prepare_builders(&*exec, &config, true) {
215                        Err(error) => {
216                            return consortium::dag::TaskOutcome::Failed(error.to_string())
217                        }
218                        Ok(None) => {}
219                        Ok(Some(path)) => {
220                            context.set_state(
221                                "machines_file",
222                                path.path().to_string_lossy().into_owned(),
223                            );
224                            let lease: std::sync::Arc<
225                                std::sync::Mutex<Option<consortium_integration::TemporaryFile>>,
226                            > = context.get_state("machines_file_lease").unwrap();
227                            *lease.lock().unwrap() = Some(path);
228                        }
229                    }
230                    consortium::dag::TaskOutcome::Success
231                }),
232            ),
233            "eval" => PhaseTask::new(
234                format!("eval:{resource}"),
235                crate::tasks::NixEvalTask::new(resource),
236            ),
237            "build" => PhaseTask::new(
238                format!("build:{resource}"),
239                crate::tasks::NixBuildTask::new(resource),
240            ),
241            "cascade" => PhaseTask::new("cascade", CascadeTask),
242            "copy" if plan.flag("cascade") => PhaseTask::new(
243                format!("copy:{resource}"),
244                CascadeCopyOutcome {
245                    host: resource.into(),
246                },
247            ),
248            "copy" => PhaseTask::new(
249                format!("copy:{resource}"),
250                crate::tasks::NixCopyTask::new(resource),
251            ),
252            "activate" => PhaseTask::new(
253                format!("activate:{resource}"),
254                crate::tasks::NixActivateTask::new(resource),
255            ),
256            _ => return Err(IntegrationError("unsupported nix phase".into())),
257        })
258    }
259}

261fn argument_words(value: &str) -> Result<Vec<String>, IntegrationError> {
262    if value.trim_start().starts_with('[') {
263        serde_json::from_str(value)
264            .map_err(|error| IntegrationError(format!("invalid argument array: {error}")))
265    } else {
266        Ok(value.split_whitespace().map(str::to_owned).collect())
267    }
268}
269
270fn local_words(value: &str) -> Result<Vec<String>, IntegrationError> {
271    if value.trim_start().starts_with('[') {
272        serde_json::from_str(value)
273            .map_err(|error| IntegrationError(format!("invalid host array: {error}")))
274    } else {
275        Ok(value.split(',').map(str::to_owned).collect())
276    }
277}

279pub(crate) fn deployment_request(
280    targets: &[String],
281    action: crate::DeployAction,
282    fanout: usize,
283    builders: bool,
284    options: &crate::DeployOptions,
285) -> crate::Result<Request> {
286    if targets.is_empty() {
287        return Err(crate::NixError::General("no target nodes selected".into()));
288    }
289    let mut request = Request::default()
290        .with("action", action.to_string())
291        .with("fanout", fanout.to_string())
292        .with("builders", builders.to_string());
293    request.targets = targets.to_vec();
294    // A typed caller may deploy a subset of the hosts its options describe
295    // (a retry pass); only the selected hosts are local for this request.
296    let local_hosts: Vec<String> = options
297        .local_hosts
298        .iter()
299        .filter(|host| targets.contains(host))
300        .cloned()
301        .collect();
302    for (key, words) in [
303        ("nixos-nix-args", &options.nix_args.nixos),
304        ("darwin-nix-args", &options.nix_args.darwin),
305        ("local", &local_hosts),
306    ] {
307        if !words.is_empty() {
308            request = request.with(
309                key,
310                serde_json::to_string(words)
311                    .map_err(|error| crate::NixError::General(error.to_string()))?,
312            );
313        }
314    }
315    if let Some(timeout) = options.activate_timeout {
316        request = request.with("activate-timeout", duration_text(timeout));
317    }
318    Ok(request)
319}

321pub(crate) fn run_deployment(
322    exec: std::sync::Arc<dyn consortium_integration::Executor>,
323    config: &FleetConfig,
324    request: &Request,
325    action: crate::DeployAction,
326) -> crate::Result<crate::DeployReport> {
327    crate::validate_targets(config, &request.targets)?;
328    let plan = NixIntegration
329        .plan(config, request)
330        .map_err(|error| crate::NixError::General(error.to_string()))?;
331    let execution = NixIntegration
332        .execute_detailed(exec, config, &plan)
333        .map_err(|error| crate::NixError::General(error.to_string()))?;
334    let mut report = crate::DeployReport::from_dag_report(&execution.dag, &plan.targets, action);
335    for (id, error) in &execution.dag.failed {
336        if !["eval:", "build:", "copy:", "activate:"]
337            .iter()
338            .any(|prefix| id.0.starts_with(prefix))
339        {
340            report.build_failures.push((id.0.clone(), error.clone()));
341        }
342    }
343    Ok(report)
344}

346struct CascadeTask;
347impl consortium::dag::DagTask for CascadeTask {
348    fn execute(&self, context: &DagContext) -> consortium::dag::TaskOutcome {
349        use consortium::dag::{TaskId, TaskOutcome};
350        use std::sync::Arc;
351        let config: FleetConfig = context.get_state("fleet_config").unwrap();
352        let plan: Plan = context.get_state("integration_plan").unwrap();
353        let options: crate::DeployOptions = context.get_state("deploy_options").unwrap();
354        let executor: Arc<dyn consortium_integration::Executor> =
355            context.get_state("executor").unwrap();
356        let mut targets = Vec::new();
357        for host in &plan.targets {
358            let Some(toplevel) = context.get_output::<String>(&TaskId(format!("build:{host}")))
359            else {
360                continue;
361            };
362            let node = &config.nodes[host];
363            if options.is_local(host) || node.build_on_target {
364                context.set_output(TaskId(format!("copy:{host}")), toplevel);
365            } else {
366                targets.push(crate::cascade_integration::CascadeCopyTarget {
367                    host_name: host.clone(),
368                    ssh_addr: crate::endpoint::ssh_address(node),
369                    toplevel_path: toplevel,
370                });
371            }
372        }
373        let sink = ExecutorEvents {
374            executor: executor.clone(),
375        };
376        let config =
377            crate::cascade_integration::CascadeCopyConfig::new(plan.value("seed"), targets.clone())
378                .fanout(plan.positive_integer("cascade-fanout").unwrap() as u32)
379                .events(&sink);
380        let result = crate::cascade_integration::cascade_copy_grouped(&executor, config);
381        for target in &targets {
382            if result.copied.contains(&target.host_name) {
383                context.set_output(
384                    TaskId(format!("copy:{}", target.host_name)),
385                    target.toplevel_path.clone(),
386                );
387            }
388        }
389        context.set_state("cascade_failures", result.failed);
390        TaskOutcome::Success
391    }
392    fn describe(&self) -> String {
393        "distribute built closures through cascade".into()
394    }
395}

397struct CascadeCopyOutcome {
398    host: String,
399}
400impl consortium::dag::DagTask for CascadeCopyOutcome {
401    fn execute(&self, context: &DagContext) -> consortium::dag::TaskOutcome {
402        if context.has_output(&consortium::dag::TaskId(format!("copy:{}", self.host))) {
403            return consortium::dag::TaskOutcome::Success;
404        }
405        let failures = context
406            .get_state::<std::collections::HashMap<String, String>>("cascade_failures")
407            .unwrap_or_default();
408        consortium::dag::TaskOutcome::Failed(
409            failures
410                .get(&self.host)
411                .cloned()
412                .unwrap_or_else(|| "cascade produced no copy output".into()),
413        )
414    }
415    fn describe(&self) -> String {
416        format!("check cascade delivery to {}", self.host)
417    }

488pub(crate) fn run_deployment_with_events(
489    exec: std::sync::Arc<dyn consortium_integration::Executor>,
490    config: &FleetConfig,
491    request: &Request,
492    action: crate::DeployAction,
493    sink: Option<&dyn crate::cascade_events::EventSink>,
494) -> crate::Result<crate::DeployReport> {
495    let Some(sink) = sink else {
496        return run_deployment(exec, config, request, action);
497    };
498    std::thread::scope(|scope| {
499        let (sender, receiver) = std::sync::mpsc::channel();
500        let worker = scope.spawn(move || {
501            run_deployment(
502                std::sync::Arc::new(EventForwarder {
503                    inner: exec,
504                    events: sender,
505                }),
506                config,
507                request,
508                action,
509            )
510        });
511        for event in receiver {
512            sink.emit(&event);
513        }
514        worker
515            .join()
516            .map_err(|_| crate::NixError::General("deployment execution panicked".into()))?
517    })
518}

736    fn assert_same_native_error(
737        expected: &crate::cascade::CascadeError,
738        actual: &crate::cascade::CascadeError,
739    ) {
740        use crate::cascade::CascadeError;
741        match (expected, actual) {
742            (
743                CascadeError::Copy {
744                    node: left,
745                    stderr: left_message,
746                },
747                CascadeError::Copy {
748                    node: right,
749                    stderr: right_message,
750                },
751            ) => {
752                assert_eq!(left, right);
753                assert_eq!(left_message, right_message);
754            }
755            (
756                CascadeError::SshHandshake {
757                    node: left,
758                    parent: left_parent,
759                },
760                CascadeError::SshHandshake {
761                    node: right,
762                    parent: right_parent,
763                },
764            ) => {
765                assert_eq!(left, right);
766                assert_eq!(left_parent, right_parent);
767            }
768            (
769                CascadeError::Activation {
770                    node: left,
771                    stage: left_stage,
772                },
773                CascadeError::Activation {
774                    node: right,
775                    stage: right_stage,
776                },
777            ) => {
778                assert_eq!(left, right);
779                assert_eq!(left_stage, right_stage);
780                assert!(std::ptr::eq(*left_stage, *right_stage));
781            }
782            (
783                CascadeError::Partitioned {
784                    src: left_src,
785                    tgt: left_tgt,
786                },
787                CascadeError::Partitioned {
788                    src: right_src,
789                    tgt: right_tgt,
790                },
791            ) => {
792                assert_eq!(left_src, right_src);
793                assert_eq!(left_tgt, right_tgt);
794            }
795            (
796                CascadeError::SubtreeAggregate {
797                    node: left,
798                    errors: left_errors,
799                },
800                CascadeError::SubtreeAggregate {
801                    node: right,
802                    errors: right_errors,
803                },
804            ) => {
805                assert_eq!(left, right);
806                assert_eq!(left_errors.len(), right_errors.len());
807                for (expected, actual) in left_errors.iter().zip(right_errors) {
808                    assert_same_native_error(expected, actual);
809                }
810            }
811            _ => panic!("native error variant changed: {expected:?} -> {actual:?}"),
812        }
813    }

815    #[test]
816    fn live_event_bridge_preserves_all_native_error_variants_and_replay_schema() {
817        use crate::cascade::{CascadeError, NodeId};
818        use crate::cascade_events::{CascadeEvent, EventSink};
819        let exec = Arc::new(ScriptedExecutor::new());
820        let (sender, receiver) = std::sync::mpsc::channel();
821        let bridge = ExecutorEvents {
822            executor: Arc::new(EventForwarder {
823                inner: exec.clone(),
824                events: sender,
825            }),
826        };
827        let cases = vec![
828            CascadeError::Copy {
829                node: NodeId(71),
830                stderr: "copy failed".into(),
831            },
832            CascadeError::SshHandshake {
833                node: NodeId(72),
834                parent: NodeId(42),
835            },
836            CascadeError::Activation {
837                node: NodeId(73),
838                stage: "switch",
839            },
840            CascadeError::Partitioned {
841                src: NodeId(74),
842                tgt: NodeId(75),
843            },
844            CascadeError::SubtreeAggregate {
845                node: NodeId(76),
846                errors: vec![
847                    CascadeError::Activation {
848                        node: NodeId(77),
849                        stage: "profile",
850                    },
851                    CascadeError::Partitioned {
852                        src: NodeId(78),
853                        tgt: NodeId(79),
854                    },
855                ],
856            },
857        ];
858        for error in cases {
859            let event = CascadeEvent::EdgeFailed {
860                round: 7,
861                src: NodeId(3),
862                tgt: NodeId(9),
863                error: error.clone(),
864            };
865            bridge.emit(&event);
866            let actual = receiver.try_recv().unwrap();
867            let CascadeEvent::EdgeFailed {
868                round,
869                src,
870                tgt,
871                error: actual_error,
872            } = &actual
873            else {
874                panic!("wrong event variant")
875            };
876            assert_eq!((*round, *src, *tgt), (7, NodeId(3), NodeId(9)));
877            assert_same_native_error(&error, actual_error);
878            assert_eq!(error.affected_nodes(), actual_error.affected_nodes());
879            let payload = serde_json::to_value(&event).unwrap();
880            assert!(payload["error"].is_string());
881            assert_eq!(serde_json::to_value(&actual).unwrap(), payload);
882            assert!(
883                matches!(exec.effects().last(), Some(Effect::Event {topic, payload: recorded}) if topic == "nix.cascade" && recorded == &payload)
884            );
885        }
886        assert_eq!(exec.effects().len(), 5);
887        assert!(receiver.try_recv().is_err());
888    }
