11/// Copy results keyed by hostname.
12pub struct CopyResults {
13    /// Hosts that were successfully copied to.
14    pub succeeded: Vec<String>,
15    /// Map of hostname -> copy error.
16    pub errors: HashMap<String, NixError>,
17}
18
19/// Copy closures to all targets in the deployment plan.
20pub fn copy_closures(exec: &dyn Executor, plan: &DeploymentPlan) -> Result<CopyResults> {
21    let mut results = CopyResults {
22        succeeded: Vec::new(),
23        errors: HashMap::new(),
24    };
25
26    // TODO: parallelize with consortium's Task/Worker fanout
27    for target in &plan.targets {
28        if !target.needs_copy {
29            results.succeeded.push(target.node.name.clone());
30            continue;
31        }
32
33        let store_uri = format!(
34            "ssh-ng://{}@{}",
35            target.node.target_user, target.node.target_host
36        );
37
38        match copy_closure_with(exec, &target.toplevel_path, &store_uri) {
39            Ok(()) => {
40                results.succeeded.push(target.node.name.clone());
41            }
42            Err(e) => {
43                results.errors.insert(target.node.name.clone(), e);
44            }
45        }
46    }
47
48    Ok(results)
49}

51/// Copy a single closure to a remote store, via an [`Executor`].
52///
53/// Uses `--no-check-sigs` because locally-built closures aren't signed
54/// by a key the remote trusts. We're deploying as root over SSH, so
55/// the trust boundary is the SSH connection itself.
56///
57/// Thin adapter over [`staging::copy_closure`] mapping errors into
58/// [`NixError`].
59pub fn copy_closure_with(exec: &dyn Executor, store_path: &str, store_uri: &str) -> Result<()> {
60    staging::copy_closure(exec, store_path, store_uri).map_err(|e| match e {
61        StagingError::CopyFailed { message, .. } => NixError::CopyFailed {
62            host: store_uri.to_string(),
63            message,
64        },
65        StagingError::Exec(source) => NixError::CopyFailed {
66            host: store_uri.to_string(),
67            message: format!("failed to run nix copy: {}", source),
68        },
69        // Unreachable from `nix copy`, mapped for exhaustiveness.
70        other => NixError::CopyFailed {
71            host: store_uri.to_string(),
72            message: other.to_string(),
73        },
74    })
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80    use consortium_integration::exec::{ExecOutput, ScriptedExecutor};
81
82    #[test]
83    fn test_copy_closure_records_nix_copy_command() {
84        let exec = ScriptedExecutor::new().on("nix copy", ExecOutput::ok(""));
85        copy_closure_with(&exec, "/nix/store/abc-env", "ssh-ng://root@hp01").unwrap();
86        exec.assert_invoked_containing(
87            "nix copy --no-check-sigs --to ssh-ng://root@hp01 /nix/store/abc-env",
88        );
89        assert_eq!(exec.invocation_count(), 1);
90    }
91
92    #[test]
93    fn test_copy_closure_failure_surfaces_stderr() {
94        let exec = ScriptedExecutor::new().on(
95            "nix copy",
96            ExecOutput::new(
97                1,
98                "",
99                "ssh: connect to host hp01 port 22: Connection refused",
100            ),
101        );
102        let err = copy_closure_with(&exec, "/nix/store/abc-env", "ssh-ng://root@hp01").unwrap_err();
103        match err {
104            NixError::CopyFailed { host, message } => {
105                assert_eq!(host, "ssh-ng://root@hp01");
106                assert!(message.contains("Connection refused"), "{}", message);
107            }
108            other => panic!("expected CopyFailed, got: {}", other),
109        }
110    }
