1//! Scheduler submission receipts are independent of the job's eventual outcome.
2
3use std::fmt::Display;
4
5use consortium::dag::{DagContext, TaskOutcome};
6use serde::Serialize;
7
8const RECEIPT_KEY: &str = "consortium_integration.submission";
9
10/// What a single submit task can establish about its request to the scheduler.
11#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
12#[non_exhaustive]
13#[serde(tag = "state", rename_all = "kebab-case")]
14pub enum SubmissionReceipt {
15    /// The scheduler acknowledged this job ID.
16    #[non_exhaustive]
17    Accepted { job_id: String },
18    /// Dispatch may have occurred, but no definitive response was available.
19    Unknown,
20    /// The scheduler explicitly refused the request.
21    Rejected,
22    /// The request demonstrably did not leave the client.
23    NotSubmitted,
24}
25
26/// Read the receipt, if a submit task actually ran in this invocation.
27///
28/// The reader for custom `DagBuilder` pipelines that run `SlurmSubmitTask`
29/// or `RaySubmitTask` directly; the integration path instead returns the
30/// receipt through `Integration::execute_with_receipt`.
31pub fn receipt(ctx: &DagContext) -> Option<SubmissionReceipt> {
32    ctx.get_state(RECEIPT_KEY)
33}
34
35/// Record an acknowledged ID; the task still provides its own typed DAG output.
36pub fn accepted(ctx: &DagContext, job_id: impl ToString) -> TaskOutcome {
37    ctx.set_state(
38        RECEIPT_KEY,
39        SubmissionReceipt::Accepted {
40            job_id: job_id.to_string(),
41        },
42    );
43    TaskOutcome::Success
44}
45
46/// Stop dependent phases when acknowledgment was lost. Include the scheduler,
47/// job name, and available evidence in `detail` so the caller can investigate.
48pub fn unknown(ctx: &DagContext, detail: impl Display) -> TaskOutcome {
49    ctx.set_state(RECEIPT_KEY, SubmissionReceipt::Unknown);
50    TaskOutcome::Failed(format!(
51        "submission unknown: a job may exist; check the scheduler before resubmitting. {detail}"
52    ))
53}
54
55/// Record an explicit scheduler rejection, not merely a nonzero exit status.
56pub fn rejected(ctx: &DagContext, detail: impl Display) -> TaskOutcome {
57    ctx.set_state(RECEIPT_KEY, SubmissionReceipt::Rejected);
58    TaskOutcome::Failed(format!("submission rejected: {detail}"))
59}
60
61/// Record a failure known to precede dispatch to the scheduler.
62pub fn not_submitted(ctx: &DagContext, detail: impl Display) -> TaskOutcome {
63    ctx.set_state(RECEIPT_KEY, SubmissionReceipt::NotSubmitted);
64    TaskOutcome::Failed(format!("not submitted: {detail}"))
65}
