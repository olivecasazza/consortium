1//! # consortium-skypilot
2//!
3//! SkyPilot multi-cloud job orchestration with nix-built environments.
4//!
5//! Launches cloud clusters via SkyPilot and tears them down afterwards,
6//! using nix for reproducible task environments. Every external command
7//! (`nix build`, `sky launch`, `sky down`) runs through an injected
8//! [`Executor`], which the DAG tasks share via the `"executor"` context
9//! state key — so the whole pipeline is testable with a
10//! [`ScriptedExecutor`](consortium_integration::exec::ScriptedExecutor)
11//! and needs neither a cloud account nor a nix installation.
12//!
13//! # Example
14//!
15//! ```
16//! use std::collections::HashMap;
17//! use std::sync::Arc;
18//!
19//! use consortium_integration::exec::{ExecOutput, Executor, ScriptedExecutor};
20//! use consortium_integration::fleet::{FleetConfig, SkypilotFleetConfig};
21//! use consortium_skypilot::{launch_task, SkyOptions};
22//!
23//! let exec: Arc<dyn Executor> = Arc::new(
24//!     ScriptedExecutor::new()
25//!         .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
26//!         .on("sky launch", ExecOutput::ok("Cluster launched\n"))
27//!         .on("sky down", ExecOutput::ok("Terminating cluster\n")),
28//! );
29//! let config = FleetConfig {
30//!     nodes: HashMap::new(),
31//!     builders: HashMap::new(),
32//!     flake_uri: ".".into(),
33//!     ansible_config: None,
34//!     slurm_config: None,
35//!     ray_config: None,
36//!     skypilot_config: Some(SkypilotFleetConfig {
37//!         cloud: "gcp".into(),
38//!         region: Some("us-central1".into()),
39//!         instance_type: None,
40//!     }),
41//! };
42//! let report = launch_task(exec, &config, "train", "resources: {}\n", &SkyOptions::default())?;
43//! assert!(report.failed.is_empty());
44//! # Ok::<(), consortium_skypilot::SkypilotError>(())
45//! ```

58/// Options for [`launch_task`].
59///
60/// The `Default` runs the full pipeline: build → launch → teardown.
61#[derive(Debug, Clone, Copy)]
62pub struct SkyOptions {
63    /// Tear the cluster down after it launches (`sky down`).
64    pub teardown: bool,
65}
66
67impl Default for SkyOptions {
68    fn default() -> Self {
69        Self { teardown: true }
70    }
71}
72
73/// Launch a SkyPilot task on a cloud cluster.
74///
75/// Pipeline: build the sky environment with nix → `sky launch` (the task
76/// yaml is written to a temp file so its path shows up in the rendered
77/// command line) → `sky down` when `opts.teardown`. Requires
78/// [`FleetConfig::skypilot_config`]; returns [`SkypilotError::NoConfig`]
79/// before issuing any command when it is absent.
80pub fn launch_task(
81    exec: Arc<dyn Executor>,
82    config: &FleetConfig,
83    cluster_name: &str,
84    task_yaml: &str,
85    opts: &SkyOptions,
86) -> Result<DagReport> {
87    let sky_config = config
88        .skypilot_config
89        .as_ref()
90        .ok_or(SkypilotError::NoConfig)?;
91
92    let ctx = DagContext::new();
93    ctx.set_state("fleet_config", config.clone());
94    ctx.set_state("executor", exec);
95
96    let mut dag = DagBuilder::new();
97
98    // Build environment
99    let build_id = format!("build-sky-env:{}", cluster_name);
100    dag.add_task(
101        &build_id,
102        tasks::NixBuildSkyEnvTask::new(cluster_name, &config.flake_uri),
103    );
104
105    // Launch cluster
106    let launch_id = format!("sky-launch:{}", cluster_name);
107    dag.add_task(
108        &launch_id,
109        tasks::SkyLaunchTask {
110            cluster_name: cluster_name.to_string(),
111            task_yaml: task_yaml.to_string(),
112            cloud: Some(sky_config.cloud.clone()),
113            region: sky_config.region.clone(),
114        },
115    );
116    dag.add_dep(&launch_id, &build_id);
117
118    // Teardown (optional)
119    if opts.teardown {
120        let down_id = format!("sky-down:{}", cluster_name);
121        dag.add_task(
122            &down_id,
123            tasks::SkyDownTask {
124                cluster_name: cluster_name.to_string(),
125            },
126        );
127        dag.add_dep(&down_id, &launch_id);
128    }
129
130    dag.error_policy(ErrorPolicy::FailFast);
131    dag.context(ctx);
132
133    let report = dag
134        .build()
135        .map_err(|e| SkypilotError::Dag(e.to_string()))?
136        .run()
137        .map_err(|e| SkypilotError::Dag(e.to_string()))?;
138
139    Ok(report)
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145    use std::collections::HashMap;
146
147    use consortium_integration::exec::{ExecOutput, ScriptedExecutor};
148    use consortium_integration::fleet::SkypilotFleetConfig;
149
150    fn config(with_skypilot: bool) -> FleetConfig {
151        FleetConfig {
152            nodes: HashMap::new(),
153            builders: HashMap::new(),
154            flake_uri: ".".into(),
155            ansible_config: None,
156            slurm_config: None,
157            ray_config: None,
158            skypilot_config: with_skypilot.then(|| SkypilotFleetConfig {
159                cloud: "gcp".into(),
160                region: Some("us-central1".into()),
161                instance_type: None,
162            }),
163        }
164    }
165
166    #[test]
167    fn launch_task_errors_without_skypilot_config() {
168        let exec: Arc<dyn Executor> = Arc::new(ScriptedExecutor::new());
169        let err =
170            launch_task(exec, &config(false), "c", "yaml", &SkyOptions::default()).unwrap_err();
171        assert!(matches!(err, SkypilotError::NoConfig));
172    }
173
174    #[test]
175    fn launch_task_happy_path_orders_phases() {
176        let scripted = Arc::new(
177            ScriptedExecutor::new()
178                .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
179                .on("sky launch", ExecOutput::ok("Cluster launched\n"))
180                .on("sky down", ExecOutput::ok("Terminating cluster\n")),
181        );
182        let exec: Arc<dyn Executor> = scripted.clone();
183        let report = launch_task(
184            exec,
185            &config(true),
186            "train",
187            "resources: {}\n",
188            &SkyOptions::default(),
189        )
190        .unwrap();
191        assert!(report.failed.is_empty(), "failures: {:?}", report.failed);
192        let build = scripted.invocation_index_containing("nix build").unwrap();
193        let launch = scripted.invocation_index_containing("sky launch").unwrap();
194        let down = scripted.invocation_index_containing("sky down").unwrap();
195        assert!(build < launch && launch < down);
196    }
197
198    #[test]
199    fn launch_task_no_teardown_skips_down() {
200        let scripted = Arc::new(
201            ScriptedExecutor::new()
202                .on("nix build", ExecOutput::ok("/nix/store/abc-sky-env\n"))
203                .on("sky launch", ExecOutput::ok("Cluster launched\n")),
204        );
205        let exec: Arc<dyn Executor> = scripted.clone();
206        let report = launch_task(
207            exec,
208            &config(true),
209            "train",
210            "resources: {}\n",
211            &SkyOptions { teardown: false },
212        )
213        .unwrap();
214        assert!(report.failed.is_empty(), "failures: {:?}", report.failed);
215        scripted.assert_not_invoked_containing("sky down");
216    }
