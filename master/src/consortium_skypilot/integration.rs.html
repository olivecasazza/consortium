1//! Adapter for the shared integration grammar.
2
3use consortium_integration::fleet::FleetConfig;
4use consortium_integration::grammar::{
5    Condition, EffectKind, FailurePolicy, Grammar, Integration, IntegrationError, Parameter,
6    ParameterKind, Phase, PhaseKind, PhaseScope, PhaseTask, Plan,
7};
8
9/// SkyPilot cluster task launch through the uniform integration contract.
10pub struct SkypilotIntegration;
11
12static GRAMMAR: Grammar = Grammar {
13    name: "skypilot",
14    about: "SkyPilot cluster task launch",
15    targets: false,
16    failure_policy: FailurePolicy::Stop,
17    parameters: &[
18        Parameter::required("name", "Cluster and Nix environment name"),
19        Parameter::required("task", "SkyPilot task YAML content"),
20        Parameter::defaulted(
21            "teardown",
22            "Tear down the cluster after launch",
23            ParameterKind::Boolean,
24            "true",
25        ),
26    ],
27    phases: &[
28        Phase::new("build").behavior(PhaseKind::Build, PhaseScope::Shared, &[EffectKind::Process]),
29        Phase::new("launch").behavior(
30            PhaseKind::Acquire,
31            PhaseScope::Shared,
32            &[EffectKind::Process, EffectKind::TemporaryFile],
33        ),
34        Phase::when("teardown", Condition::Equals("teardown", "true")).behavior(
35            PhaseKind::Release,
36            PhaseScope::Shared,
37            &[EffectKind::Process],
38        ),
39    ],
40};
41
42impl Integration for SkypilotIntegration {
43    fn cleanup(&self, context: &consortium::dag::DagContext) -> Result<(), IntegrationError> {
44        if let Some(name) = context.get_state::<String>("sky-cluster-name") {
45            crate::tasks::cleanup_task_yaml(context, &name).map_err(|error| {
46                IntegrationError(format!("temporary file cleanup failed: {error}"))
47            })?;
48        }
49        Ok(())
50    }
51
52    fn configure(
53        &self,
54        _config: &FleetConfig,
55        plan: &Plan,
56        context: &consortium::dag::DagContext,
57    ) -> Result<(), IntegrationError> {
58        context.set_state("sky-cluster-name", plan.value("name").to_owned());
59        Ok(())
60    }
61
62    fn grammar(&self) -> &'static Grammar {
63        &GRAMMAR
64    }
65
66    fn validate_config(&self, config: &FleetConfig) -> Result<(), IntegrationError> {
67        if config.skypilot_config.is_none() {
68            return Err(IntegrationError(
69                "skypilot: missing fleet configuration".into(),
70            ));
71        }
72        Ok(())
73    }
74
75    fn task(
76        &self,
77        config: &FleetConfig,
78        plan: &Plan,
79        phase: &Phase,
80        _resource: &str,
81    ) -> Result<PhaseTask, IntegrationError> {
82        let c = config
83            .skypilot_config
84            .as_ref()
85            .ok_or_else(|| IntegrationError("skypilot: missing configuration".into()))?;
86        let name = plan.value("name");
87        Ok(match phase.name {
88            "build" => PhaseTask::new(
89                format!("build-sky-env:{name}"),
90                crate::tasks::NixBuildSkyEnvTask::new(name, &config.flake_uri),
91            ),
92            "launch" => PhaseTask::new(
93                format!("sky-launch:{name}"),
94                crate::tasks::SkyLaunchTask {
95                    cluster_name: name.into(),
96                    task_yaml: plan.value("task").into(),
97                    cloud: Some(c.cloud.clone()),
98                    region: c.region.clone(),
99                },
100            ),
101            "teardown" => PhaseTask::new(
102                format!("sky-down:{name}"),
103                crate::tasks::SkyDownTask {
104                    cluster_name: name.into(),
105                },
106            ),
107            _ => return Err(IntegrationError("unsupported skypilot phase".into())),
108        })
109    }
110}
