1//! DagTask implementations for SkyPilot orchestration.
2//!
3//! Every task reads the shared `Arc<dyn Executor>` from the DAG context
4//! (state key `"executor"`, set by `launch_task()`) and fails with
5//! `executor not in context` when the key is absent. All commands run
6//! locally: the nix environment build goes through
7//! [`staging::build_flake_attr`], and the SkyPilot CLI commands
8//! (`sky launch` / `sky exec` / `sky down`) run against the operator's
9//! local cloud credentials.
10
11use std::sync::{Arc, Mutex};
12
13use consortium::dag::{DagContext, DagTask, TaskId, TaskOutcome};
14use consortium_integration::exec::{CommandSpec, Executor};
15use consortium_integration::staging;
16
17type TaskFile = Arc<Mutex<Option<consortium_integration::TemporaryFile>>>;
18
19/// Release invocation files after resource teardown has had a chance to run.
20pub fn cleanup_task_yaml(ctx: &DagContext, cluster_name: &str) -> std::io::Result<()> {
21    if let Some(file) = ctx.get_state::<TaskFile>(&format!("sky-task-file:{cluster_name}")) {
22        if let Some(file) = file.lock().unwrap().take() {
23            file.close()?;
24        }
25    }
26    Ok(())
27}
28
29/// Fetch the shared executor from the DAG context.
30fn executor_from(ctx: &DagContext) -> Result<Arc<dyn Executor>, TaskOutcome> {
31    ctx.get_state::<Arc<dyn Executor>>("executor")
32        .ok_or_else(|| TaskOutcome::Failed("executor not in context".into()))
33}

35/// Build a skypilot task environment via nix.
36pub struct NixBuildSkyEnvTask {
37    pub env_name: String,
38    pub flake_attr: String,
39}
40
41impl NixBuildSkyEnvTask {
42    pub fn new(env_name: &str, flake_uri: &str) -> Self {
43        Self {
44            env_name: env_name.to_string(),
45            flake_attr: format!("{}#skyEnvs.{}", flake_uri, env_name),
46        }
47    }
48}
49
50impl DagTask for NixBuildSkyEnvTask {
51    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
52        let exec = match executor_from(ctx) {
53            Ok(e) => e,
54            Err(outcome) => return outcome,
55        };
56
57        match staging::build_flake_attr(&*exec, &self.flake_attr, None) {
58            Ok(path) => {
59                ctx.set_output(TaskId(format!("build-sky-env:{}", self.env_name)), path);
60                TaskOutcome::Success
61            }
62            Err(e) => TaskOutcome::Failed(format!("build sky env: {}", e)),
63        }
64    }
65
66    fn describe(&self) -> String {
67        format!("build skypilot environment '{}'", self.env_name)
68    }
69}

71/// Launch a SkyPilot cluster.
72pub struct SkyLaunchTask {
73    pub cluster_name: String,
74    pub task_yaml: String,
75    pub cloud: Option<String>,
76    pub region: Option<String>,
77}
78
79impl DagTask for SkyLaunchTask {
80    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
81        let exec = match executor_from(ctx) {
82            Ok(e) => e,
83            Err(outcome) => return outcome,
84        };
85
86        let yaml_path = match exec.temporary_file("consortium-sky", self.task_yaml.as_bytes()) {
87            Ok(p) => p,
88            Err(e) => return TaskOutcome::Failed(format!("sky launch failed: {}", e)),
89        };
90
91        let mut spec = CommandSpec::new("sky").args([
92            "launch".to_string(),
93            "-c".to_string(),
94            self.cluster_name.clone(),
95            yaml_path.path().to_string_lossy().into_owned(),
96            "-y".to_string(),
97        ]);
98        if let Some(ref cloud) = self.cloud {
99            spec = spec.args(["--cloud".to_string(), cloud.clone()]);
100        }
101        if let Some(ref region) = self.region {
102            spec = spec.args(["--region".to_string(), region.clone()]);
103        }
104
105        ctx.set_state(
106            &format!("sky-task-file:{}", self.cluster_name),
107            Arc::new(Mutex::new(Some(yaml_path))),
108        );
109        match exec.exec(&spec) {
110            Ok(out) if out.success() => {
111                ctx.set_output(
112                    TaskId(format!("sky-launch:{}", self.cluster_name)),
113                    self.cluster_name.clone(),
114                );
115                TaskOutcome::Success
116            }
117            Ok(out) => TaskOutcome::Failed(format!("sky launch failed: {}", out.stderr.trim())),
118            Err(e) => TaskOutcome::Failed(format!("sky launch failed: {}", e)),
119        }
120    }
121
122    fn describe(&self) -> String {
123        format!("launch sky cluster '{}'", self.cluster_name)
124    }
125}

127/// Execute a command on a SkyPilot cluster.
128pub struct SkyExecTask {
129    pub cluster_name: String,
130    pub command: String,
131}
132
133impl DagTask for SkyExecTask {
134    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
135        let exec = match executor_from(ctx) {
136            Ok(e) => e,
137            Err(outcome) => return outcome,
138        };
139
140        let spec = CommandSpec::new("sky").args([
141            "exec".to_string(),
142            self.cluster_name.clone(),
143            "--".to_string(),
144            self.command.clone(),
145        ]);
146
147        match exec.exec(&spec) {
148            Ok(out) if out.success() => {
149                ctx.set_output(
150                    TaskId(format!("sky-exec:{}", self.cluster_name)),
151                    out.stdout,
152                );
153                TaskOutcome::Success
154            }
155            Ok(out) => TaskOutcome::Failed(format!("sky exec failed: {}", out.stderr.trim())),
156            Err(e) => TaskOutcome::Failed(format!("sky exec failed: {}", e)),
157        }
158    }
159
160    fn describe(&self) -> String {
161        format!("exec on sky cluster '{}'", self.cluster_name)
162    }
163}

188    }

361    #[test]
362    fn yaml_is_scoped_around_launch_for_success_and_failure() {
363        for output in [ExecOutput::ok("launched"), ExecOutput::new(1, "", "denied")] {
364            let exec = Arc::new(ScriptedExecutor::new().on("sky launch", output.clone()));
365            let result = launch(exec.clone());
366            assert_eq!(matches!(result, TaskOutcome::Success), output.success());
367            assert!(
368                matches!(&exec.effects()[..], [Effect::TemporaryFileCreated { path, contents }, Effect::Command { spec, .. }, Effect::TemporaryFileRemoved { path: removed }] if contents == b"resources: {}" && path == removed && spec.args.iter().any(|arg| std::path::Path::new(arg) == path))
369            );
370        }
371    }

398    #[test]
399    fn cleanup_errors_do_not_change_the_acquisition_outcome() {
400        struct CleanupFailure {
401            command_fails: bool,
402        }
403        impl Executor for CleanupFailure {
404            fn exec(&self, _: &CommandSpec) -> Result<ExecOutput, ExecError> {
405                if self.command_fails {
406                    Err(ExecError::Scripted("command failed".into()))
407                } else {
408                    Ok(ExecOutput::ok("launched"))
409                }
410            }
411            fn temporary_file(&self, _: &str, _: &[u8]) -> std::io::Result<TemporaryFile> {
412                Ok(TemporaryFile::new("/virtual/task".into(), |_| {
413                    Err(std::io::Error::new(
414                        std::io::ErrorKind::PermissionDenied,
415                        "cleanup denied",
416                    ))
417                }))
418            }
419        }
420        for command_fails in [false, true] {
421            let ctx = DagContext::new();
422            ctx.set_state(
423                "executor",
424                Arc::new(CleanupFailure { command_fails }) as Arc<dyn Executor>,
425            );
426            let result = SkyLaunchTask {
427                cluster_name: "test".into(),
428                task_yaml: "resources: {}".into(),
429                cloud: None,
430                region: None,
431            }
432            .execute(&ctx);
433            assert_eq!(result.is_success(), !command_fails);
434            assert_eq!(
435                ctx.has_output(&TaskId("sky-launch:test".into())),
436                !command_fails
437            );
438            assert!(cleanup_task_yaml(&ctx, "test")
439                .unwrap_err()
440                .to_string()
441                .contains("cleanup denied"));
442        }
443    }
