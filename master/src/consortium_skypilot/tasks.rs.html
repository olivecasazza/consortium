1//! DagTask implementations for SkyPilot orchestration.
2//!
3//! Every task reads the shared `Arc<dyn Executor>` from the DAG context
4//! (state key `"executor"`, set by `launch_task()`) and fails with
5//! `executor not in context` when the key is absent. All commands run
6//! locally: the nix environment build goes through
7//! [`staging::build_flake_attr`], and the SkyPilot CLI commands
8//! (`sky launch` / `sky exec` / `sky down`) run against the operator's
9//! local cloud credentials.
10
11use std::path::PathBuf;
12use std::sync::atomic::{AtomicU64, Ordering};
13use std::sync::Arc;
14
15use consortium::dag::{DagContext, DagTask, TaskId, TaskOutcome};
16use consortium_integration::exec::{CommandSpec, Executor};
17use consortium_integration::staging;

19/// Fetch the shared executor from the DAG context.
20fn executor_from(ctx: &DagContext) -> Result<Arc<dyn Executor>, TaskOutcome> {
21    ctx.get_state::<Arc<dyn Executor>>("executor")
22        .ok_or_else(|| TaskOutcome::Failed("executor not in context".into()))
23}
24
25/// Write a SkyPilot task yaml to a unique temp file and return its path.
26///
27/// Purely local fs, no executor: `sky launch` takes a yaml *path*, and the
28/// path then appears in the rendered command line, keeping the invocation
29/// matchable for scripted executors.
30fn write_task_yaml(cluster_name: &str, task_yaml: &str) -> std::io::Result<PathBuf> {
31    static SEQ: AtomicU64 = AtomicU64::new(0);
32    let seq = SEQ.fetch_add(1, Ordering::Relaxed);
33    let path = std::env::temp_dir().join(format!(
34        "consortium-sky-{}-{}-{}.yaml",
35        cluster_name,
36        std::process::id(),
37        seq
38    ));
39    std::fs::write(&path, task_yaml)?;
40    Ok(path)
41}

43/// Build a skypilot task environment via nix.
44pub struct NixBuildSkyEnvTask {
45    pub env_name: String,
46    pub flake_attr: String,
47}
48
49impl NixBuildSkyEnvTask {
50    pub fn new(env_name: &str, flake_uri: &str) -> Self {
51        Self {
52            env_name: env_name.to_string(),
53            flake_attr: format!("{}#skyEnvs.{}", flake_uri, env_name),
54        }
55    }
56}
57
58impl DagTask for NixBuildSkyEnvTask {
59    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
60        let exec = match executor_from(ctx) {
61            Ok(e) => e,
62            Err(outcome) => return outcome,
63        };
64
65        match staging::build_flake_attr(&*exec, &self.flake_attr, None) {
66            Ok(path) => {
67                ctx.set_output(TaskId(format!("build-sky-env:{}", self.env_name)), path);
68                TaskOutcome::Success
69            }
70            Err(e) => TaskOutcome::Failed(format!("build sky env: {}", e)),
71        }
72    }
73
74    fn describe(&self) -> String {
75        format!("build skypilot environment '{}'", self.env_name)
76    }
77}

79/// Launch a SkyPilot cluster.
80pub struct SkyLaunchTask {
81    pub cluster_name: String,
82    pub task_yaml: String,
83    pub cloud: Option<String>,
84    pub region: Option<String>,
85}
86
87impl DagTask for SkyLaunchTask {
88    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
89        let exec = match executor_from(ctx) {
90            Ok(e) => e,
91            Err(outcome) => return outcome,
92        };
93
94        let yaml_path = match write_task_yaml(&self.cluster_name, &self.task_yaml) {
95            Ok(p) => p,
96            Err(e) => return TaskOutcome::Failed(format!("sky launch failed: {}", e)),
97        };
98
99        let mut spec = CommandSpec::new("sky").args([
100            "launch".to_string(),
101            "-c".to_string(),
102            self.cluster_name.clone(),
103            yaml_path.to_string_lossy().into_owned(),
104            "-y".to_string(),
105        ]);
106        if let Some(ref cloud) = self.cloud {
107            spec = spec.args(["--cloud".to_string(), cloud.clone()]);
108        }
109        if let Some(ref region) = self.region {
110            spec = spec.args(["--region".to_string(), region.clone()]);
111        }
112
113        match exec.exec(&spec) {
114            Ok(out) if out.success() => {
115                ctx.set_output(
116                    TaskId(format!("sky-launch:{}", self.cluster_name)),
117                    self.cluster_name.clone(),
118                );
119                TaskOutcome::Success
120            }
121            Ok(out) => TaskOutcome::Failed(format!("sky launch failed: {}", out.stderr.trim())),
122            Err(e) => TaskOutcome::Failed(format!("sky launch failed: {}", e)),
123        }
124    }
125
126    fn describe(&self) -> String {
127        format!("launch sky cluster '{}'", self.cluster_name)
128    }
129}

131/// Execute a command on a SkyPilot cluster.
132pub struct SkyExecTask {
133    pub cluster_name: String,
134    pub command: String,
135}
136
137impl DagTask for SkyExecTask {
138    fn execute(&self, ctx: &DagContext) -> TaskOutcome {
139        let exec = match executor_from(ctx) {
140            Ok(e) => e,
141            Err(outcome) => return outcome,
142        };
143
144        let spec = CommandSpec::new("sky").args([
145            "exec".to_string(),
146            self.cluster_name.clone(),
147            "--".to_string(),
148            self.command.clone(),
149        ]);
150
151        match exec.exec(&spec) {
152            Ok(out) if out.success() => {
153                ctx.set_output(
154                    TaskId(format!("sky-exec:{}", self.cluster_name)),
155                    out.stdout,
156                );
157                TaskOutcome::Success
158            }
159            Ok(out) => TaskOutcome::Failed(format!("sky exec failed: {}", out.stderr.trim())),
160            Err(e) => TaskOutcome::Failed(format!("sky exec failed: {}", e)),
161        }
162    }
163
164    fn describe(&self) -> String {
165        format!("exec on sky cluster '{}'", self.cluster_name)
166    }
167}
