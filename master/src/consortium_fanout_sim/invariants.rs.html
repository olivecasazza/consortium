1//! Reusable cascade invariant assertions.
2//!
3//! Extracted from `tests/fuzz.rs` so the fuzz harness, the curated
4//! corpus (`tests/corpus.rs`), and later simulator stages all assert
5//! the same contract with the same messages. Every assertion panics
6//! with the strategy name and scenario seed so a failure can be
7//! replayed without digging through logs.
8//!
9//! These are plain `assert!`-based helpers (not proptest's
10//! `prop_assert!`): a violated invariant is a hard failure either way,
11//! and panics keep them usable from both `proptest!` blocks and plain
12//! `#[test]` corpus tests.
13
14use std::collections::HashSet;

16use consortium_nix::cascade::{CascadeResult, CascadeStrategy, NodeId};
17
18use crate::scenario::{Scenario, ScenarioConfig};

20/// Every node converged and the run reported success.
21pub fn assert_converged_all(strategy: &str, seed: u64, result: &CascadeResult, n_nodes: u32) {
22    assert!(
23        result.is_success(),
24        "[{strategy}] seed={seed:#x} failed unexpectedly: {:?}",
25        result.failed,
26    );
27    assert_eq!(
28        result.converged.len() as u32,
29        n_nodes,
30        "[{strategy}] seed={seed:#x} not all nodes converged ({} of {n_nodes})",
31        result.converged.len(),
32    );
33}

35/// The converged list contains no duplicates.
36pub fn assert_no_duplicate_converged(strategy: &str, seed: u64, result: &CascadeResult) {
37    let converged_set: HashSet<NodeId> = result.converged.iter().copied().collect();
38    assert_eq!(
39        converged_set.len(),
40        result.converged.len(),
41        "[{strategy}] seed={seed:#x} duplicates in converged list",
42    );
43}

45/// Round count stays within the coordinator's `max_rounds` bound and
46/// `round_durations` has exactly one entry per round.
47///
48/// The bound is deliberately `max_rounds`, not `n_nodes`: since the
49/// coordinator stopped marking targets permanently failed on transient
50/// edge failures, an unlucky failure pattern can legitimately exceed
51/// `n_nodes` rounds while the strategy retries alternate sources.
52pub fn assert_round_bounds(strategy: &str, seed: u64, result: &CascadeResult, max_rounds: u32) {
53    assert!(
54        result.rounds <= max_rounds,
55        "[{strategy}] seed={seed:#x} rounds {} > max_rounds {max_rounds}",
56        result.rounds,
57    );
58    assert_eq!(
59        result.round_durations.len() as u32,
60        result.rounds,
61        "[{strategy}] seed={seed:#x} round_durations len mismatch",
62    );
63}

65/// Seed-aware semantics for [`crate::fixtures::FailureSchedule::KillNodeAtRound`]
66/// injected at round 0.
67///
68/// The kill schedule fails edges whose TARGET is the killed node
69/// (receive-side death). Whether the killed node must fail therefore
70/// depends on whether it was pre-seeded:
71///
72/// - `killed ∈ seeded`: the node starts converged and never needs an
73///   inbound copy, so the kill can never fire. It MUST remain in
74///   `converged` (and keeps serving as a source), and no failure is
75///   required in the result.
76/// - `killed ∉ seeded`: every attempt to deliver the closure to it
77///   fails from round 0. It MUST NOT appear in `converged`, and
78///   `result.failed` MUST name it in `affected_nodes()`.
79///
80/// The first branch is what the pre-2025 fuzz oracle got wrong: it
81/// asserted "killed ∉ converged" unconditionally, which is
82/// wrong-by-construction whenever the draw picks the pre-seeded node
83/// (`NodeId(0)` under `SeedDistribution::Single`) — the root cause of
84/// the historical "killed node still appears in converged set" flake.
85pub fn assert_killed_node_semantics(
86    strategy: &str,
87    seed: u64,
88    result: &CascadeResult,
89    killed: NodeId,
90    seeded: &HashSet<NodeId>,
91) {
92    if seeded.contains(&killed) {
93        assert!(
94            result.converged.contains(&killed),
95            "[{strategy}] seed={seed:#x} pre-seeded node {killed:?} lost the closure",
96        );
97    } else {
98        assert!(
99            !result.converged.contains(&killed),
100            "[{strategy}] seed={seed:#x} killed node {killed:?} still appears in converged \
101             set: {:?}",
102            result.converged,
103        );
104        let err = result.failed.as_ref().unwrap_or_else(|| {
105            panic!(
106                "[{strategy}] seed={seed:#x} killed node injected at round 0 but \
107                 result.failed is None"
108            )
109        });
110        assert!(
111            err.affected_nodes().contains(&killed),
112            "[{strategy}] seed={seed:#x} killed node {killed:?} missing from affected set",
113        );
114    }
115}

117/// Running the same scenario twice with the same strategy must produce
118/// identical round counts, per-round durations, and converged sets.
119///
120/// Compares full converged SETS, not just counts: same count with a
121/// different set would mean the cascade makes non-deterministic edge
122/// choices that a `len ==` check would never notice.
123pub fn assert_deterministic(cfg: &ScenarioConfig, strategy: &dyn CascadeStrategy) {
124    let r1 = Scenario::new(cfg.clone()).run(strategy);
125    let r2 = Scenario::new(cfg.clone()).run(strategy);
126    assert_eq!(
127        r1.rounds,
128        r2.rounds,
129        "[{}] seed={:#x} round count differs between identical runs",
130        strategy.name(),
131        cfg.seed,
132    );
133    assert_eq!(
134        r1.round_durations,
135        r2.round_durations,
136        "[{}] seed={:#x} round durations differ between identical runs",
137        strategy.name(),
138        cfg.seed,
139    );
140    let s1: HashSet<NodeId> = r1.converged.iter().copied().collect();
141    let s2: HashSet<NodeId> = r2.converged.iter().copied().collect();
142    assert_eq!(
143        s1,
144        s2,
145        "[{}] seed={:#x} converged sets diverge between identical-seed runs",
146        strategy.name(),
147        cfg.seed,
148    );
149}
