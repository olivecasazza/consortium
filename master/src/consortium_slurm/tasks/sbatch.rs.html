1//! Interpret the default (non-`--parsable`) sbatch protocol, not arbitrary SSH output.
2
3use consortium_integration::exec::ExecOutput;
4
5const SUBMITTED: &str = "Submitted batch job ";
6const FAILED: &str = "sbatch: error: Batch job submission failed: ";
7const PARTITION_DETAIL: &str = "sbatch: error: invalid partition specified: ";
8
9/// OpenSSH's first-contact notice wraps an arbitrary `<host>` (<keytype>)
10/// between a fixed prefix and suffix.
11const SSH_NOTICE_PREFIX: &str = "Warning: Permanently added ";
12const SSH_NOTICE_SUFFIX: &str = " to the list of known hosts.";
13
14/// The wait task can address a single unqualified numeric job ID on the submit node.
15/// Upstream sbatch prints `Submitted batch job %u` followed by a newline. Without
16/// the terminator, even valid-looking digits might be the prefix of a different ID.
17pub(super) fn acknowledged_job_id(output: &ExecOutput) -> Option<u64> {
18    if !output.success() {
19        return None;
20    }
21    let line = output.stdout.strip_suffix('\n')?;
22    let digits = line
23        .strip_suffix('\r')
24        .unwrap_or(line)
25        .strip_prefix(SUBMITTED)?;
26    if !matches!(digits.as_bytes(), [b'1'..=b'9', ..]) {
27        return None;
28    }
29    let id = digits.parse::<u32>().ok()?;
30    // A host-key warning is not a contradiction; a second receipt or batch failure is.
31    if output.stderr.lines().any(|line| {
32        line.starts_with(SUBMITTED)
33            || line.starts_with(FAILED)
34            || line.starts_with(PARTITION_DETAIL)
35    }) {
36        return None;
37    }
38    Some(u64::from(id))
39}
40
41/// Recognize the controller's invalid-partition refusal, optionally preceded by
42/// its partition detail. Require sbatch's default error status as well: SSH loss,
43/// a signal, or a timeout is not a scheduler refusal. OpenSSH first-contact
44/// host-key notices are not part of the refusal and are ignored.
45pub(super) fn rejected_invalid_partition(output: &ExecOutput) -> bool {
46    if output.status != 1 || !output.stdout.is_empty() || !output.stderr.ends_with('\n') {
47        return false;
48    }
49    let mut lines = output
50        .stderr
51        .lines()
52        .filter(|line| !(line.starts_with(SSH_NOTICE_PREFIX) && line.ends_with(SSH_NOTICE_SUFFIX)))
53        .rev();
54    let reason = lines.next().and_then(|line| line.strip_prefix(FAILED));
55    if reason != Some("Invalid partition name specified") {
56        return false;
57    }
58    match lines.next() {
59        None => true,
60        Some(detail) => {
61            detail
62                .strip_prefix(PARTITION_DETAIL)
63                .is_some_and(|partition| !partition.is_empty())
64                && lines.next().is_none()
65        }
66    }
67}
