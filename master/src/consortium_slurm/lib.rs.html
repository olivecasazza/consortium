1//! # consortium-slurm
2//!
3//! Slurm job orchestration with nix-built environments.
4//!
5//! Uses nix to build hermetic job environments, copies them to the submit
6//! node, then submits jobs with slurm. Supports arbitrary DAG pipelines
7//! (e.g., RNA-seq bioinformatics workflows) via DagBuilder.
8//!
9//! All external commands run through an injected
10//! `consortium_integration::exec::Executor`: production callers pass a
11//! `ProcessExecutor`, tests pass a `ScriptedExecutor`.
12
13pub mod error;
14pub mod tasks;
15
16pub use error::{Result, SlurmError};

18use std::sync::Arc;
19use std::time::Duration;

21use consortium::dag::{DagBuilder, DagContext, DagReport, ErrorPolicy};
22use consortium_integration::exec::Executor;

25/// Options for [`submit_job`].
26///
27/// The default exercises the full pipeline — build → copy → submit → wait —
28/// with no result collection.
29#[derive(Debug, Clone)]
30pub struct SubmitOptions {
31    /// Wait for the job to reach a terminal state (polls sacct).
32    /// Default: `true`.
33    pub wait: bool,
34    /// Glob pattern of output files to `cat` on the submit node once the job
35    /// has finished (e.g. `"slurm-12345.out"`). Default: `None` (no
36    /// collection).
37    pub collect: Option<String>,
38    /// sacct poll interval used while waiting. Default: 10s.
39    pub poll_interval: Duration,
40}
41
42impl Default for SubmitOptions {
43    fn default() -> Self {
44        Self {
45            wait: true,
46            collect: None,
47            poll_interval: Duration::from_secs(10),
48        }
49    }
50}

52/// Submit a single slurm job with a nix-built environment.
53///
54/// Runs the pipeline `build-job-env → copy-to-submit → submit → wait →
55/// collect` as a DAG: the wait stage is included when
56/// [`SubmitOptions::wait`] is set, and the collect stage when
57/// [`SubmitOptions::collect`] is a pattern — wired after `slurm-wait` when
58/// waiting is enabled, directly after `slurm-submit` otherwise. All commands
59/// run through `exec`.
60///
61/// Returns [`SlurmError::NoConfig`] before any command is issued when the
62/// fleet config has no slurm sub-config.
63pub fn submit_job(
64    exec: Arc<dyn Executor>,
65    config: &FleetConfig,
66    job_name: &str,
67    script: &str,
68    partition: Option<&str>,
69    opts: &SubmitOptions,
70) -> Result<DagReport> {
71    let slurm_config = config.slurm_config.as_ref().ok_or(SlurmError::NoConfig)?;
72
73    let ctx = DagContext::new();
74    ctx.set_state("fleet_config", config.clone());
75    ctx.set_state("executor", exec);
76
77    let mut dag = DagBuilder::new();
78
79    // Build job environment
80    let build_id = format!("build-job-env:{job_name}");
81    dag.add_task(
82        &build_id,
83        tasks::NixBuildJobEnvTask::new(job_name, &config.flake_uri),
84    );
85
86    // Copy to submit node
87    let copy_id = format!("copy-job-env:{job_name}");
88    dag.add_task(
89        &copy_id,
90        tasks::NixCopyToSubmitTask {
91            job_name: job_name.to_string(),
92            submit_host: slurm_config.submit_node.clone(),
93            submit_user: slurm_config.submit_user.clone(),
94        },
95    );
96    dag.add_dep(&copy_id, &build_id);
97
98    // Submit
99    let submit_id = format!("slurm-submit:{job_name}");
100    dag.add_task(
101        &submit_id,
102        tasks::SlurmSubmitTask {
103            job_name: job_name.to_string(),
104            script: script.to_string(),
105            partition: partition.map(|s| s.to_string()),
106            submit_host: slurm_config.submit_node.clone(),
107            submit_user: slurm_config.submit_user.clone(),
108        },
109    );
110    dag.add_dep(&submit_id, &copy_id);
111
112    // Wait (optional) — collect hangs off whichever stage ran last.
113    let mut last_id = submit_id.clone();
114    if opts.wait {
115        let wait_id = format!("slurm-wait:{job_name}");
116        dag.add_task(
117            &wait_id,
118            tasks::SlurmWaitTask::new(
119                job_name,
120                &slurm_config.submit_node,
121                &slurm_config.submit_user,
122            )
123            .with_poll_interval(opts.poll_interval),
124        );
125        dag.add_dep(&wait_id, &submit_id);
126        last_id = wait_id;
127    }
128
129    // Collect (optional)
130    if let Some(ref pattern) = opts.collect {
131        let collect_id = format!("slurm-collect:{job_name}");
132        dag.add_task(
133            &collect_id,
134            tasks::SlurmCollectTask {
135                job_name: job_name.to_string(),
136                output_pattern: pattern.clone(),
137                submit_host: slurm_config.submit_node.clone(),
138                submit_user: slurm_config.submit_user.clone(),
139            },
140        );
141        dag.add_dep(&collect_id, &last_id);
142    }
143
144    dag.error_policy(ErrorPolicy::FailFast);
145    dag.context(ctx);
146
147    let report = dag
148        .build()
149        .map_err(|e| SlurmError::Dag(e.to_string()))?
150        .run()
151        .map_err(|e| SlurmError::Dag(e.to_string()))?;
152
153    Ok(report)
154}
